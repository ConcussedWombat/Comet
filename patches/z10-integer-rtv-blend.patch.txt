+    static ON: std::sync::OnceLock<bool> = std::sync::OnceLock::new();

+fn z10_integer_blend(
+    mut blend: D3D12_BLEND_DESC,
+    targets: &D3D12_RT_FORMAT_ARRAY,
+) -> (D3D12_BLEND_DESC, u8) {
+    let independent = blend.IndependentBlendEnable.as_bool();
+    let mut changed = 0u8;
+    for (i, &format) in targets.RTFormats.iter()
+        .take(targets.NumRenderTargets as usize).enumerate()
+    {
+        let effective = if independent { i } else { 0 };
+        if z10_integer_rtv(format) && blend.RenderTarget[effective].BlendEnable.as_bool() {
+            changed |= 1 << i;
+        }
+    }
+    if changed != 0 {
+        if !independent {
+            // Common state applies to every target; preserve it before splitting.
+            blend.RenderTarget = [blend.RenderTarget[0]; 8];
+            blend.IndependentBlendEnable = true.into();
+        }
+        for (i, entry) in blend.RenderTarget.iter_mut().enumerate() {
+            if changed & (1 << i) != 0 {
+                entry.BlendEnable = false.into();
+            }
+        }
+    }
+    (blend, changed)
+}
