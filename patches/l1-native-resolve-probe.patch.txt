+// Inspect the original output immediately after the real dispatch, before
+// the B1 clear or any later consumer can overwrite it. This adds a copy and
+// transitions, so it is a diagnostic run, not a timing or ordering control.
+#[cfg(feature = "diagnostics")]
+pub(crate) fn l1_probe_resolve(list: &HostList, target: &Target) {
+    static COUNT: AtomicU64 = AtomicU64::new(0);
+    let d = unsafe { target.texture.GetDesc() };
+    if d.Width.saturating_mul(u64::from(d.Height)) > 4_194_304 {
+        return;
+    }
+    let Ok(id) = COUNT.fetch_update(Ordering::Relaxed, Ordering::Relaxed,
+        |n| (n < 16).then_some(n + 1)) else { return };
+    let Some(host) = crate::host::host() else { return };
+    let title = target.title as crate::object::This;
+    list.keep(&target.texture);
+    crate::list::ensure_state(list, title, D3D12_RESOURCE_STATE_COPY_SOURCE.0 as u32);
+    let read = crate::capture::read_back(&host.device, &list.list, &target.texture);
+    crate::list::ensure_state(list, title, D3D12_RESOURCE_STATE_UNORDERED_ACCESS.0 as u32);
+    if let Some((buffer, footprint)) = read {
+        let frame = crate::queue::presented() + 1;
+        list.note_op(|| format!("L1 RESOLVE PROBE recorded id={id} hash={} title={:#x}", target.hash, target.title));
+        list.defer_l1_resolve(ResolveReadback {
+            buffer, footprint, frame, hash: target.hash.clone(), title: target.title, id,
+        });
+    }
+}

+#[cfg(feature = "diagnostics")]
+impl ResolveReadback {
+    /// Invoked after the host list finishes, not at command recording.
+    pub(crate) fn report(&self) {
+        let f = self.footprint.Footprint;
+        let width = u64::from(f.Width);
+        let height = u64::from(f.Height);
+        let pitch = u64::from(f.RowPitch);
+        let offset = self.footprint.Offset;
+        let allocation = unsafe { self.buffer.GetDesc() }.Width;
+        if f.Format != DXGI_FORMAT_R32_UINT || f.Depth != 1 || width == 0 || height == 0
+            || width * 4 > pitch
+            || offset.checked_add((height - 1) * pitch).and_then(|n| n.checked_add(width * 4))
+                .is_none_or(|end| end > allocation)
+        {
+            tracing::warn!(id = self.id, "L1 RESOLVE PROBE invalid readback footprint");
+            return;
+        }
+        let mut data = std::ptr::null_mut();
+        // SAFETY: readback is complete; every row is within the allocation.
+        unsafe {
+            if self.buffer.Map(0, None, Some(&mut data)).is_err() || data.is_null() {
+                tracing::warn!(id = self.id, "L1 RESOLVE PROBE map failed");
+                return;
+            }
+            let mut zero_rgb = 0u64;
+            let mut all_ones = 0u64;
+            let mut peak = 0.0f32;
+            let mut sum = 0.0f64;
+            for y in 0..height {
+                let row = std::slice::from_raw_parts(data.cast::<u8>().add((offset + y * pitch) as usize), (width * 4) as usize);
+                for b in row.chunks_exact(4) {
+                    let v = u32::from_le_bytes([b[0], b[1], b[2], b[3]]);
+                    zero_rgb += u64::from(v & 0x07ff_ffff == 0);
+                    all_ones += u64::from(v == u32::MAX);
+                    let scale = 2f32.powi((v >> 27) as i32 - 24);
+                    for shift in [0, 9, 18] {
+                        let c = ((v >> shift) & 511) as f32 * scale;
+                        peak = peak.max(c);
+                        sum += f64::from(c);
+                    }
+                }
+            }
+            self.buffer.Unmap(0, None);
+            tracing::info!(id = self.id, frame = self.frame, hash = %self.hash,
+                title = format_args!("{:#x}", self.title), width, height, zero_rgb, all_ones,
+                peak, mean_rgb = sum / (3 * width * height) as f64,
+                "L1 RESOLVE PROBE native packed RGB9E5 after dispatch");
+        }
+    }
+}
