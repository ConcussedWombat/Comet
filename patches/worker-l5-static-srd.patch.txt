+//! RDNA2 ISA Buffer Resource Descriptor fields: address bits 47:0,
+//! stride bits 61:48, NUM_RECORDS bits 95:64 (in units of nonzero stride).
+//! Cache/swizzle and word 3 are not determined by a structured view.
+pub fn l5_structured_srd_word(
+    address: u64, stride: u32, records: u32, word: u32, low30: bool,
+) -> Option<u32> {
+    if address >> 48 != 0 || stride == 0 || stride >= 1 << 14 {
+        return None;
+    }
+    match word {
+        0 => Some(address as u32),
+        1 if low30 => Some((address >> 32) as u32 | stride << 16),
+        2 => Some(records),
+        _ => None,
+    }
