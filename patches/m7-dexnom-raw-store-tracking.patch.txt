+#[cfg(windows)]
+fn raw_instruction_may_write(encoding: comet_shader::gfx10::Encoding, name: Option<&str>) -> bool {
+    use comet_shader::gfx10::Encoding;
+    let Some(name) = name else { return true };
+    match encoding {
+        // Do not classify other flat variants as read-only.
+        Encoding::Flat | Encoding::Global => !name.starts_with("global_load_"),
+        Encoding::Smem => !name.starts_with("s_load_dword") && !name.starts_with("s_buffer_load_dword"),
+        Encoding::Sop1 => matches!(name, "s_setpc_b64" | "s_swappc_b64"),
+        _ => false,
+    }
+}
