+#[test]
+fn l4_scc_edge_requires_wide_and_matching_exec() {
+    for skip in [0xBF88_0000, 0xBF89_0000, 0xBF84_0000, 0xBF85_0000] {
+        let native = program(skip);
+        let p = ir::build(&native.code, native.wave64, native.vgprs).expect("ir::build");
+        let (b, block) = p.blocks.iter().enumerate().find(|(_, b)| {
+            b.insts.last().is_some_and(|i| i.name().starts_with("s_cbranch"))
+        }).expect("branch");
+        let scc = block.insts.iter().find(|i| i.name() == "s_and_saveexec_b64")
+            .and_then(|i| i.write(ir::Reg::Scc)).expect("SCC");
+        let ir::Exit::Branch { taken, not_taken } = block.exit else { panic!("branch") };
+        let narrow = structure::structure(&p).expect("structure");
+        assert_eq!(lower::l4_scc_edge_fact(&p, &narrow, scc, b, taken), None);
+        let wide = structure::structure_wide(&p, &HashSet::from([b])).expect("wide");
+        let taken_bit = matches!(skip, 0xBF89_0000 | 0xBF85_0000);
+        assert_eq!(lower::l4_scc_edge_fact(&p, &wide, scc, b, taken), Some(taken_bit));
+        assert_eq!(lower::l4_scc_edge_fact(&p, &wide, scc, b, not_taken), Some(!taken_bit));
+        // A different value, even if mentioned by the same terminator,
+        // is not the result of the same tracked SCC reduction.
+        let other = block.insts.iter().find(|i| i.name() == "s_and_saveexec_b64")
+            .and_then(|i| i.write(ir::Reg::S(12))).expect("saved EXEC");
+        assert_eq!(lower::l4_scc_edge_fact(&p, &wide, other, b, taken), None);
+    }
