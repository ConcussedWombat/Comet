+    /// Only direct, constant scalar masks/extracts prove that some bits
+    /// are unobservable. A phi, move or any other use demands every bit.
+    fn l5_srd_demand(&self, value: Value) -> u32 {
+        let mut mask = 0;
+        for block in &self.x.p.blocks {
+            if block.phis.iter().any(|p| p.args.contains(&value)) {
+                return u32::MAX;
+            }
+            for i in &block.insts {
+                if !i.reads.iter().any(|r| r.1 == value) {
+                    continue;
+                }
+                let Fields::Sop { src, .. } = &i.fields else {
+                    return u32::MAX;
+                };
+                let is_value = |op: &Operand| {
+                    matches!(*op, Operand::Scalar { code, dwords: 1 }
+                        if i.read(Reg::S(code)) == Some(value))
+                };
+                let constant = |op: &Operand| match *op {
+                    Operand::Literal(x) => Some(x),
+                    Operand::Const(c) => crate::resource::inline(c),
+                    _ => None,
+                };
+                let (Some(a), Some(b)) = (src.first(), src.get(1)) else {
+                    return u32::MAX;
+                };
+                let bits = match i.name() {
+                    "s_and_b32" if is_value(a) => constant(b),
+                    "s_and_b32" if is_value(b) => constant(a),
+                    "s_bfe_u32" | "s_bfe_i32" if is_value(a) => constant(b).map(|x| {
+                        let offset = x & 31;
+                        let width = ((x >> 16) & 127).min(32 - offset);
+                        if width == 32 {
+                            u32::MAX
+                        } else {
+                            ((1u32 << width) - 1) << offset
+                        }
+                    }),
+                    _ => None,
+                };
+                let Some(bits) = bits else { return u32::MAX };
+                mask |= bits;
+            }
+        }
+        mask
+    }

+    fn l5_root_index(&self, value: Value) -> Option<(u8, u8, u8)> {

+        let Fields::Sop { src, .. } = &i.fields else { return None; };
+        let Operand::Scalar { code, dwords: 1 } = *src.first()? else { return None; };

+//! RDNA2 ISA Buffer Resource Descriptor fields: address bits 47:0,
+//! stride bits 61:48, NUM_RECORDS bits 95:64 (in units of nonzero stride).
+//! Cache/swizzle and word 3 are not determined by a structured view.
+pub fn l5_structured_srd_word(
+    address: u64, stride: u32, records: u32, word: u32, low30: bool,
+) -> Option<u32> {
+    if address >> 48 != 0 || stride == 0 || stride >= 1 << 14 {
+        return None;
+    }
+    match word {
+        0 => Some(address as u32),
+        1 if low30 => Some((address >> 32) as u32 | stride << 16),
+        2 => Some(records),
+        _ => None,
+    }
+}

+/// Draw-evaluable descriptor address encoded in RootType::Other.
+/// UserData::dword is its constant BYTE displacement.
+#[derive(Clone, Copy, Debug, PartialEq, Eq)]
+pub struct L5SrdIndex {
+    pub heap: bool,
+    pub low30: bool,
+    pub constant: Option<(u8, u8, u8)>, // slot, dword, shift
+}
+impl L5SrdIndex {
+    const PREFIX: u32 = 0x5100_0000;
+    pub fn encode(self) -> u32 {
+        let mut kind = Self::PREFIX | (u32::from(self.heap) << 23)
+            | (u32::from(self.low30) << 22);
+        if let Some((slot, dword, shift)) = self.constant {
+            kind |= (1 << 21) | (u32::from(shift & 31) << 16)
+                | (u32::from(dword) << 8) | u32::from(slot);
+        }
+        kind
+    }
+    pub fn decode(kind: u32) -> Option<Self> {
+        if kind & 0xff00_0000 != Self::PREFIX { return None; }
+        Some(Self {
+            heap: kind & (1 << 23) != 0,
+            low30: kind & (1 << 22) != 0,
+            constant: (kind & (1 << 21) != 0)
+                .then_some((kind as u8, (kind >> 8) as u8, ((kind >> 16) & 31) as u8)),
+        })
+    }
+    pub fn byte_offset(self, base: u32, offset: u32, constant: u32) -> Option<u32> {
+        base.checked_add(offset)?.checked_add(self.constant
+            .map_or(0, |(_, _, shift)| constant.wrapping_shl(u32::from(shift))))
+    }
+}
