+/// GPU wins per byte if it differs from the pre-submission snapshot;
+/// otherwise take the title's current byte. Return whether title changes.
+/// Inputs are a single 64-byte line. No alignment assumption.
+#[cfg(target_arch = "x86_64")]
+fn p2_merge_line(gpu: &[u8; 64], old: &[u8; 64], title: &[u8; 64]) -> ([u8; 64], bool) {
+    use std::arch::x86_64::*;
+    let mut result = [0u8; 64];
+    let mut changed = false;
+    for at in (0..64).step_by(16) {
+        // SAFETY: x86_64 includes SSE2; each unaligned load/store covers
+        // exactly 16 bytes of its 64-byte array. The arrays remain live.
+        unsafe {
+            let g = _mm_loadu_si128(gpu.as_ptr().add(at).cast());
+            let o = _mm_loadu_si128(old.as_ptr().add(at).cast());
+            let t = _mm_loadu_si128(title.as_ptr().add(at).cast());
+            let keep_cpu = _mm_cmpeq_epi8(g, o);
+            let merged = _mm_or_si128(_mm_and_si128(keep_cpu, t), _mm_andnot_si128(keep_cpu, g));
+            changed |= _mm_movemask_epi8(_mm_cmpeq_epi8(merged, t)) != 0xffff;
+            _mm_storeu_si128(result.as_mut_ptr().add(at).cast(), merged);
+        }
+    }
+    (result, changed)
+}

+fn p2_simd_merge_line_matches_bytewise() {
+    // Each of the 64 positions sees every combination of unchanged,
+    // CPU-only, GPU-only, same write and conflicting writes.
+    for phase in 0..7 {
+        let old = std::array::from_fn(|i| (i * 13 + 7) as u8);
+        let mut gpu = old;
+        let mut title = old;
+        for i in 0..64 {
+            match (i + phase) % 7 {
+                0 => {}
+                1 => gpu[i] ^= 0x80,
+                2 => title[i] ^= 0x01,
+                3 => { gpu[i] ^= 0xff; title[i] ^= 0xff; }
+                4 => { gpu[i] ^= 0xff; title[i] ^= 0x01; }
+                5 => { gpu[i] ^= 0x01; title[i] ^= 0x80; }
+                _ => { gpu[i] ^= 0x7f; title[i] ^= 0x7e; }
+            }
+        }
+        let expected: [u8; 64] = std::array::from_fn(|i| if gpu[i] != old[i] { gpu[i] } else { title[i] });
+        assert_eq!(p2_merge_line(&gpu, &old, &title), (expected, expected != title));
+    }
+    let old = [0u8; 64];
+    let mut gpu = old;
+    let mut title = old;
+    // Both sides changed the same byte; title already contains the answer.
+    gpu[63] = 255;
+    title[63] = 255;
+    assert_eq!(p2_merge_line(&gpu, &old, &title), (gpu, false));
+}
