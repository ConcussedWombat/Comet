+//! Z13: record original PSO descriptions in an opt-in, versioned local book,
+//! then replay used keys offline through Comet's normal shader/PSO cache.
+//! Based on avatarmanz/VIctory-7900XTX b7ab9c4 and a30829c (GPL-3.0-or-later;
+//! VIctory by Marvisak and contributors; fork changes by avatarmanz).
+use std::io::{self, Read, Write};
+use std::path::{Path, PathBuf};
+use std::sync::atomic::{AtomicU64, Ordering};
+use std::sync::{Arc, OnceLock};
+use std::time::Instant;
+
+use super::{Shader, host::{GraphicsState, HostPipeline}, library};
+
+const MAGIC: &[u8] = b"COMETZ13";
+const STAGES: [&str; 8] = ["vertex", "pixel", "domain", "hull", "geometry", "compute", "amplification", "mesh"];
+const MAX_SHADER: usize = 64 * 1024 * 1024;
+const MAX_ENTRY: usize = 3 * 1024 * 1024;
+
+fn folder() -> Option<PathBuf> {
+    Some(library::warm_folder()?.join("z13-book-v1"))
+}
+
+fn read_limited(path: &Path, max: usize) -> Option<Vec<u8>> {
+    let mut out = Vec::new();
+    std::fs::File::open(path).ok()?.take(max as u64 + 1).read_to_end(&mut out).ok()?;
+    (out.len() <= max).then_some(out)
+}
+
+// Independent, immutable files make interrupted recording recoverable.
+// Publish shader files before the descriptor referring to them.
+fn publish(path: &Path, bytes: &[u8]) -> io::Result<()> {
+    if path.is_file() { return Ok(()) }
+    let parent = path.parent().ok_or(io::Error::other("no parent"))?;
+    std::fs::create_dir_all(parent)?;
+    static NEXT: AtomicU64 = AtomicU64::new(0);
+    let temp = parent.join(format!(".z13-{}-{}.tmp", std::process::id(), NEXT.fetch_add(1, Ordering::Relaxed)));
+    let result = (|| {
+        let mut f = std::fs::OpenOptions::new().write(true).create_new(true).open(&temp)?;
+        f.write_all(bytes)?;
+        drop(f);
+        std::fs::rename(&temp, path).or_else(|e| if path.is_file() { Ok(()) } else { Err(e) })
+    })();
+    let _ = std::fs::remove_file(temp);
+    result
+}
+
+pub(super) fn record(key: &str, shaders: &[Arc<Shader>], graphics: Option<&GraphicsState>) {
+    if !library::warm_enabled() || !library::valid_warm_key(key) || shaders.is_empty() || shaders.len() > 8 {
+        return;
+    }
+    let Some(dir) = folder() else { return };
+    let path = dir.join("psos").join(key);
+    if path.is_file() { return; }
+    let mut entry = MAGIC.to_vec();
+    entry.push(shaders.len() as u8);
+    for shader in shaders {
+        if shader.container.is_empty() || shader.container.len() > MAX_SHADER { return; }
+        let Some(stage) = STAGES.iter().position(|s| *s == shader.stage) else { return };
+        let hash = blake3::hash(&shader.container).to_hex().to_string();
+        let bytes = dir.join("shaders").join(&hash);
+        if let Err(error) = publish(&bytes, &shader.container) {
+            tracing::warn!(%error, %key, "Z13 shader book write failed");
+            return;
+        }
+        entry.push(stage as u8);
+        entry.extend(hash.as_bytes());
+    }
+    match graphics {
+        None => entry.push(0),
+        Some(g) => {
+            let Some(state) = g.z13_encode() else { return };
+            entry.push(1);
+            entry.extend(state);
+        }
+    }
+    if entry.len() > MAX_ENTRY { return; }
+    if let Err(error) = publish(&path, &entry) {
+        tracing::warn!(%error, %key, "Z13 PSO book write failed");
+    }
+}
+
+fn decode(entry: &[u8], mut read: impl FnMut(&str) -> Option<Vec<u8>>)
+    -> Option<(Vec<Arc<Shader>>, Option<Box<GraphicsState>>)> {
+    let (&count, mut rest) = entry.strip_prefix(MAGIC)?.split_first()?;
+    if count == 0 || count > 8 { return None; }
+    let mut shaders = Vec::with_capacity(count as usize);
+    for _ in 0..count {
+        let (raw, tail) = rest.split_first_chunk::<65>()?;
+        rest = tail;
+        let stage = *STAGES.get(raw[0] as usize)?;
+        let hash = std::str::from_utf8(&raw[1..]).ok()?;
+        if !hash.bytes().all(|b| b.is_ascii_hexdigit()) { return None; }
+        let container = read(hash)?;
+        if container.is_empty() || container.len() > MAX_SHADER
+            || blake3::hash(&container).to_hex().as_str() != hash { return None; }
+        // Do not intern offline replay: each PSO's translations can drop
+        // immediately, rather than retaining every shader across the book.
+        shaders.push(Arc::new(Shader {
+            stage, hash: hash[..16].to_owned(), container,
+            ambient_resolve: OnceLock::new(), translated: OnceLock::new(),
+        }));
+    }
+    let (&tag, rest) = rest.split_first()?;
+    let state = match tag {
+        0 if rest.is_empty() => None,
+        1 => Some(Box::new(GraphicsState::z13_decode(rest)?)),
+        _ => return None,
+    };
+    Some((shaders, state))
+}
+
+pub(super) fn replay(offset: usize, limit: usize) -> io::Result<()> {
+    if !library::warm_enabled() {
+        return Err(io::Error::other("COMET_WARM=1 required"));
+    }
+    let dir = folder().ok_or(io::Error::other("pipeline cache not configured"))?;
+    // Select only previously requested keys with complete descriptions.
+    let keys = library::warm_keys();
+    let began = Instant::now();
+    let (mut made, mut skipped, mut refused) = (0u64, 0u64, 0u64);
+    tracing::info!(known_used = keys.len(), offset, limit, "Z13 warm start");
+    for key in keys.iter().skip(offset).take(limit) {
+        let Some((shaders, g)) = read_limited(&dir.join("psos").join(key), MAX_ENTRY)
+            .and_then(|entry| decode(&entry, |hash| read_limited(&dir.join("shaders").join(hash), MAX_SHADER)))
+        else {
+            skipped += 1;
+            continue;
+        };
+        // Same original descriptor and normal D3D12 pipeline library;
+        // a stale host blob is rejected by library::open as on a normal run.
+        tracing::info!(%key, "Z13 warm compiling");
+        let started = Instant::now();
+        let ok = HostPipeline::new(&shaders, g.as_deref()).is_some();
+        let millis = started.elapsed().as_millis() as u64;
+        library::warm_note_time(key, millis);
+        made += u64::from(ok);
+        refused += u64::from(!ok);
+        tracing::info!(%key, millis, ok, "Z13 warm PSO");
+    }
+    library::save(true);
+    tracing::info!(made, refused, skipped, elapsed_ms = began.elapsed().as_millis() as u64, "Z13 warm done");
+    Ok(())
+}
