+enum State {
+    #[default]
+    Fresh,
+    Pending,
+    Ready(u64, u64, u64), // address, bytes, fingerprint of the actual snapshot
+    Listed,
+    Failed, // submission may still own the resource; never show or reuse it

+#[derive(Default)]
+pub(crate) struct Gate {
+    state: Mutex<State>,
+    landed: Condvar,
+}
+
+impl Gate {
+    fn get(&self) -> State {
+        *self.state.lock().unwrap_or_else(|e| e.into_inner())
+    }
+
+    fn claim(&self) -> bool {
+        let mut state = self.state.lock().unwrap_or_else(|e| e.into_inner());
+        if *state != State::Fresh { return false; }
+        *state = State::Pending;
+        true
+    }
+
+    fn finish(&self, next: State) {
+        let mut state = self.state.lock().unwrap_or_else(|e| e.into_inner());
+        debug_assert_eq!(*state, State::Pending);
+        *state = next;
+        self.landed.notify_all();
+    }
+
+    fn before_use(&self, list: bool) -> bool {
+        let mut state = self.state.lock().unwrap_or_else(|e| e.into_inner());
+        while *state == State::Pending {
+            state = self.landed.wait(state).unwrap_or_else(|e| e.into_inner());
+        }
+        if *state == State::Failed { return false; }
+        if list || *state == State::Fresh { *state = State::Listed; }
+        true
+    }
+
+    pub(crate) fn before_descriptor(&self) -> bool { self.before_use(false) }
+    pub(crate) fn before_list(&self) -> bool { self.before_use(true) }
+}

+fn reserve(bytes: u64) -> bool {
+    BYTES.fetch_update(Ordering::AcqRel, Ordering::Acquire, |old| {
+        old.checked_add(bytes).filter(|&n| n <= LIMIT)
+    }).is_ok()
+}

+pub(crate) fn take(resource: This, whole: bool) -> Taken {
+    if !on() { return Taken::Listed; }
+    let Some(copy) = crate::resource::host_copy(resource) else { return Taken::Listed };
+    let state = copy.ahead.get();
+    if state != State::Fresh { return taken(state); }
+    // The copy queue relies on COMMON promotion/decay. No alias, no host
+    // writes, no partial resident-mip job, no texture already used by a list.
+    if !whole || BROKEN.load(Ordering::Acquire) || copy.state.load(Ordering::Acquire) != 0
+        || !super::title_written(&copy.texture) || super::host_wrote(resource as usize)
+        || crate::resource::alias_of(resource).is_some() || super::is_staged(resource)
+        || workers().is_none()
+    { return Taken::Listed; }
+    if !copy.ahead.claim() { return taken(copy.ahead.get()); }
+    let Some(staging) = super::build_staging(resource, &copy.texture) else {
+        copy.ahead.finish(State::Listed);
+        return Taken::Listed;
+    };
+    let read = staging.read_len as u64;
+    let charged = read.saturating_add(staging.bytes);
+    if read == 0 || !reserve(charged) {
+        copy.ahead.finish(State::Listed);
+        return Taken::Listed;
+    }
+    let mut queue = QUEUE.0.lock().unwrap_or_else(|e| e.into_inner());
+    if BROKEN.load(Ordering::Acquire) {
+        BYTES.fetch_sub(charged, Ordering::AcqRel);
+        copy.ahead.finish(State::Listed);
+        return Taken::Listed;
+    }
+    queue.push_back(Job { copy, staging, charged, asked: std::time::Instant::now() });
+    QUEUE.1.notify_one();
+    Taken::Queued(read)
+}

+/// Same windows as texture::fingerprint, but of the bytes actually copied.
+/// A later title write must not be mistaken for the uploaded snapshot.
+fn print(bytes: &[u8]) -> u64 {
+    const W: usize = 16 << 10;
+    let n = bytes.len();
+    let mut hash = blake3::Hasher::new();
+    if n <= 3 * W {
+        hash.update(bytes);
+    } else {
+        for start in [0, (n / 2) & !0xfff, n - W] {
+            hash.update(&bytes[start..start + W]);
+        }
+    }
+    let mut first = [0; 8];
+    first.copy_from_slice(&hash.finalize().as_bytes()[..8]);
+    u64::from_le_bytes(first) | 1
+}

+    static STARTED: OnceLock<Option<()>> = OnceLock::new();
+    *STARTED.get_or_init(|| {
+        let host = crate::host::host()?;
+        let device = &host.device;
+        // SAFETY: creating a dedicated COPY queue and private worker objects.
+        let queue: ID3D12CommandQueue = unsafe {
+            device.CreateCommandQueue(&D3D12_COMMAND_QUEUE_DESC {
+                Type: D3D12_COMMAND_LIST_TYPE_COPY, ..Default::default()
+            }).ok()?
+        };
+        let first = Worker::new(device, queue.clone()).ok()?;
+        let second = Worker::new(device, queue).ok()?;
+        for (n, worker) in [first, second].into_iter().enumerate() {
+            std::thread::Builder::new().name(format!("comet-z6-upload-{n}"))
+                .spawn(move || worker.run()).ok()?;
+        }
+        tracing::info!("Z6 upload-ahead enabled: 2 copy workers, 96 MiB cap");
+        Some(())
+    })
+}
+
+impl Worker {
+    fn new(device: &ID3D12Device, queue: ID3D12CommandQueue) -> windows::core::Result<Self> {
+        // SAFETY: private allocator and list; reset only after its fence.
+        unsafe {
+            let allocator: ID3D12CommandAllocator = device.CreateCommandAllocator(D3D12_COMMAND_LIST_TYPE_COPY)?;
+            let list: ID3D12GraphicsCommandList = device.CreateCommandList(0, D3D12_COMMAND_LIST_TYPE_COPY, &allocator, None)?;
+            list.Close()?;
+            Ok(Self { queue, allocator, list, fence: device.CreateFence(0, D3D12_FENCE_FLAG_NONE)? })
+        }
+    }
+
+    fn run(self) {
+        let mut value = 0u64;
+        loop {
+            let mut jobs = Vec::new();
+            {
+                let mut queue = QUEUE.0.lock().unwrap_or_else(|e| e.into_inner());
+                while queue.is_empty() && !BROKEN.load(Ordering::Acquire) {
+                    queue = QUEUE.1.wait(queue).unwrap_or_else(|e| e.into_inner());
+                }
+                if BROKEN.load(Ordering::Acquire) { return; }
+                let mut bytes = 0;
+                while jobs.len() < 64 {
+                    let Some(next) = queue.front() else { break };
+                    if !jobs.is_empty() && bytes + next.charged > BATCH { break; }
+                    let job = queue.pop_front().unwrap();
+                    bytes += job.charged;
+                    jobs.push(job);
+                }
+            }
+            let mut ready = Vec::new();
+            for job in jobs {
+                if let Some(state) = job.prepare() {
+                    ready.push((job, state));
+                } else {
+                    tracing::debug!("Z6 snapshot incomplete; ordinary upload will retry");
+                    job.finish(State::Listed);
+                }
+            }
+            if ready.is_empty() { continue; }
+            value += 1;
+            let mut submitted = false;
+            let result = self.copy(&ready, value, &mut submitted);
+            if result.is_err() {
+                tracing::error!(?result, submitted, "Z6 copy queue failed; disabling ahead");
+                disable();
+                for (job, _) in ready {
+                    if submitted {
+                        // Fence is unproved: neither return pooled memory nor
+                        // reset the allocator nor publish this resource.
+                        job.copy.ahead.finish(State::Failed);
+                        std::mem::forget(job);
+                    } else {
+                        job.finish(State::Listed);
+                    }
+                }
+                if submitted { std::mem::forget(self); }
+                return;
+            }
+            for (job, state) in ready {
+                let n = FINISHED.fetch_add(1, Ordering::Relaxed) + 1;
+                if n == 1 || n.is_multiple_of(500) {
+                    tracing::info!(n, bytes = job.staging.read_len, elapsed_ms = job.asked.elapsed().as_millis() as u64,
+                        inflight = BYTES.load(Ordering::Relaxed), "Z6 upload-ahead fence landed");
+                }
+                job.finish(state);
+            }
+        }
+    }
+
+    fn copy(&self, jobs: &[(Job, State)], value: u64, submitted: &mut bool) -> windows::core::Result<()> {
+        // SAFETY: an exclusive allocator/list, COMMON fresh resources and
+        // upload buffers held until a confirmed fence. Copy queues decay to
+        // COMMON after execution; graphics state tracking remains unchanged.
+        unsafe {
+            self.allocator.Reset()?;
+            self.list.Reset(&self.allocator, None)?;
+            for (job, _) in jobs {
+                for (i, footprint) in job.staging.footprints.iter().enumerate() {
+                    super::record_subresource(&self.list, &job.copy.texture, &job.staging, i, footprint);
+                }
+            }
+            self.list.Close()?;
+            let list: ID3D12CommandList = self.list.cast()?;
+            self.queue.ExecuteCommandLists(&[Some(list)]);
+            *submitted = true;
+            self.queue.Signal(&self.fence, value)?;
+            self.fence.SetEventOnCompletion(value, windows::Win32::Foundation::HANDLE::default())?;
+            let completed = self.fence.GetCompletedValue();
+            if completed == u64::MAX || completed < value {
+                return Err(windows::core::Error::from_hresult(windows::core::HRESULT(0x80004005u32 as i32)));
+            }
+        }
+        Ok(())
+    }
+}
+
+fn disable() {
+    let mut queue = QUEUE.0.lock().unwrap_or_else(|e| e.into_inner());
+    BROKEN.store(true, Ordering::Release);
+    let abandoned = std::mem::take(&mut *queue);
+    QUEUE.1.notify_all();
+    drop(queue);
+    for job in abandoned { job.finish(State::Listed); }
+}

+#[cfg(test)]
+mod tests {
+    use super::*;
+
+    #[test]
+    fn first_claim_is_exclusive_and_ready_only_after_finish() {
+        let gate = Arc::new(Gate::default());
+        assert!(gate.claim());
+        assert!(!gate.claim());
+        assert_eq!(gate.get(), State::Pending);
+        let (tx, rx) = std::sync::mpsc::channel();
+        let other = gate.clone();
+        let worker = std::thread::spawn(move || tx.send(other.before_descriptor()).unwrap());
+        assert!(rx.try_recv().is_err());
+        gate.finish(State::Ready(4096, 16, 7));
+        assert!(rx.recv().unwrap());
+        worker.join().unwrap();
+        assert_eq!(gate.get(), State::Ready(4096, 16, 7));
+        assert!(gate.before_list());
+        assert_eq!(gate.get(), State::Listed);
+        assert!(!gate.claim());
+    }
+
+    #[test]
+    fn prior_descriptor_or_list_prevents_claim_and_failed_never_shows() {
+        let gate = Gate::default();
+        assert!(gate.before_descriptor());
+        assert!(!gate.claim());
+        let gate = Gate::default();
+        assert!(gate.claim());
+        gate.finish(State::Failed);
+        assert!(!gate.before_descriptor());
+        assert!(!gate.before_list());
+    }
+
+    #[test]
+    fn fingerprint_changes_with_uploaded_bytes() {
+        let mut bytes = vec![0; 64 * 1024];
+        let original = print(&bytes);
+        bytes[0] = 1;
+        assert_ne!(print(&bytes), original);
+        assert_eq!(print(&bytes) & 1, 1);
+    }
