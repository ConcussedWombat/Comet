+            if let Ok(c) = comet_shader::Container::parse(&s.container)

+                && let Ok(programs) = comet_shader::native_programs(part.data)
+            {
+                for program in programs {
+                    used.extend(program.used.into_iter().map(|u| (s.hash.clone(), u)));
+                }
+            }
+        }
+    }
