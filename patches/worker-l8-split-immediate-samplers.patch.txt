+/// One GFX10 sampler can be used by both comparison and ordinary image
+/// instructions. DXIL's sampler kind is fixed for each register.
+fn immediate_sampler_index(
+    samplers: &mut Vec<[u32; 4]>,
+    comparisons: &mut u64,
+    words: [u32; 4],
+    compare: bool,
+) -> Result<usize, Why> {
+    if let Some(n) = samplers.iter().enumerate().position(|(n, w)| {
+        *w == words && (*comparisons >> n & 1 != 0) == compare
+    }) {
+        return Ok(n);
+    }
+    let n = samplers.len();
+    if n >= u64::BITS as usize {
+        return Err("more than 64 immediate sampler bindings");
+    }
+    samplers.push(words);
+    if compare {
+        *comparisons |= 1 << n;
+    }
+    Ok(n)
+}

+#[cfg(test)]
+#[test]
+fn immediate_sampler_kind_is_part_of_its_identity() {
+    let words = [1, 2, 3, 4];
+    for first in [false, true] {
+        let mut samplers = Vec::new();
+        let mut comparisons = 0;
+        let a = immediate_sampler_index(&mut samplers, &mut comparisons, words, first).unwrap();
+        let b = immediate_sampler_index(&mut samplers, &mut comparisons, words, !first).unwrap();
+        assert_eq!((a, b), (0, 1));
+        assert_eq!(samplers, vec![words, words]);
+        assert_eq!(comparisons, if first { 1 } else { 2 });
+        assert_eq!(
+            immediate_sampler_index(&mut samplers, &mut comparisons, words, first),
+            Ok(a)
+        );
+        assert_eq!(
+            immediate_sampler_index(&mut samplers, &mut comparisons, words, !first),
+            Ok(b)
+        );
+    }
+}

+    let mut samplers = Vec::new();

+            Ok(n as usize)

+    }

+        Ok(63)

+    assert_eq!(comparisons, 1u64 << 63);

+        immediate_sampler_index(&mut samplers, &mut comparisons, words, true),
+        Ok(63)
+    );
+    assert!(immediate_sampler_index(&mut samplers, &mut comparisons, words, false).is_err());
+    assert_eq!(samplers.len(), 64);
+    assert_eq!(comparisons, 1u64 << 63);
+}
