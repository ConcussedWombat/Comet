+//! VICTORY_THIN (3a99ee6); GPL-3.0, VIctory by Marvisak and contributors.
+//! Unlike that fork's separate back-list and async submission lane, Comet

+impl Call {
+    fn run(self, this: This) {
+        match self {
+            Self::Close => {
+                if let Some(host) = host_list_raw(this) {
+                    host.close();
+                }
+            }
+            Self::Indirect { kind, stride, max, address, counter, signature, argument, counted } => {
+                INDIRECT.fetch_add(1, Relaxed);
+                let mut ran = false;
+                with_state_now(this, |list, s| {
+                    ran = s.execute_indirect(list, (kind, stride, max), address, counter);
+                });
+                if !ran {
+                    indirect_dropped(signature, stride, &[argument], max, Some(address), counted);
+                }
+            }
+            #[cfg(test)]
+            Self::Test(values, value) => lock(&values).push(value),
+            call => with_state_now(this, |list, s| match call {
+                Self::Pipeline(p) => s.set_pipeline(p as This),
+                Self::Topology(v) => s.set_topology(list, v),
+                Self::Viewports(v) => s.set_viewports(list, &v),
+                Self::Scissors(v) => s.set_scissors(list, &v),
+                Self::Blend(v) => s.set_blend_factor(list, v.as_ref()),
+                Self::Stencil(v) => s.set_stencil_ref(list, v),
+                Self::DepthBounds(min, max) => s.set_depth_bounds(list, min, max),
+                Self::Table(c, slot, h) => s.set_table(c, slot, h),
+                Self::Constant(c, slot, v, offset) => s.set_constants(c, slot, &[v], offset),
+                Self::Constants(c, slot, v, offset) => s.set_constants(c, slot, &v, offset),
+                Self::View(c, slot, a) => s.set_view(c, slot, a),
+                Self::Index(v) => s.set_index_buffer(v.as_ref()),
+                Self::Vertices(start, count, v) => match v {
+                    Some(v) => s.set_vertex_buffers(start, &v),
+                    None => s.clear_vertex_buffers(start, count),
+                },
+                Self::Draw(a, b, c, d) => s.draw(list, a, b, c, d),
+                Self::Indexed(a, b, c, d, e) => s.draw_indexed(list, a, b, c, d, e),

+#[derive(Default)]
+struct Pool {
+    ready: Mutex<VecDeque<(usize, Arc<Journal>)>>,
+    work: Condvar,

+
+static POOL: OnceLock<Option<Arc<Pool>>> = OnceLock::new();
+static QUEUED: AtomicU64 = AtomicU64::new(0);
+static REPLAYED: AtomicU64 = AtomicU64::new(0);
+static BATCHES: AtomicU64 = AtomicU64::new(0);
+static INDIRECT: AtomicU64 = AtomicU64::new(0);
+static WORK_NS: AtomicU64 = AtomicU64::new(0);
+static WAIT_NS: AtomicU64 = AtomicU64::new(0);

+fn lock<T>(mutex: &Mutex<T>) -> MutexGuard<'_, T> {
+    mutex.lock().unwrap_or_else(|e| e.into_inner())
+}
+
+fn start() -> Option<Arc<Pool>> {
+    if !std::env::var_os("COMET_THIN").is_some_and(|v| v == "1")
+        || crate::host::host().is_none()
+    {
+        return None;
+    }
+    let pool = Arc::new(Pool::default());
+    let mut started = 0;
+    for i in 0..WORKERS {
+        let p = pool.clone();
+        match std::thread::Builder::new().name(format!("comet-list-{i}")).spawn(move || {
+            loop {
+                let (address, journal) = {
+                    let mut ready = lock(&p.ready);
+                    while ready.is_empty() {
+                        ready = p.work.wait(ready).unwrap_or_else(|e| e.into_inner());
+                    }
+                    ready.pop_front().expect("nonempty ready queue")
+                };
+                // A panic cannot publish a partial recording or leave a
+                // submission waiting forever.
+                if std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
+                    journal.run(address as This);
+                })).is_err() {
+                    crate::host::fail(format_args!("COMET_THIN list translator panicked"));
+                }
+            }
+        }) {
+            Ok(_) => started += 1,
+            Err(e) => tracing::warn!(error = %e, "COMET_THIN could not start a translator"),
+        }
+    }
+    if started == 0 {
+        return None; // original path, no journal allocated
+    }
+    tracing::info!(started, chunk = CHUNK, "COMET_THIN list translator pool");
+    Some(pool)
+}

+pub(super) fn new_journal() -> Option<Arc<Journal>> {
+    POOL.get_or_init(start).as_ref().map(|_| Arc::new(Journal::default()))
+}
+
+fn pool() -> Option<&'static Pool> {
+    POOL.get().and_then(Option::as_deref)
+}
+
+fn journal(this: This) -> Option<&'static Arc<Journal>> {
+    if this.is_null() || pool().is_none() {
+        return None;
+    }
+    // SAFETY: title list methods pass a live List. The original host list
+    // remains owned by that object; release/reset drain before mutation.
+    unsafe { (*this.cast::<List>()).thin.as_ref() }
+}
+
+fn publish(p: &Pool, this: This, journal: &Arc<Journal>, pending: &mut Pending) {
+    if !pending.active && !pending.calls.is_empty() {
+        pending.active = true;
+        lock(&p.ready).push_back((this as usize, journal.clone()));
+        p.work.notify_one();
+    }
+}
+
+pub(super) fn record(this: This, make: impl FnOnce() -> Call) -> bool {
+    let (Some(p), Some(j)) = (pool(), journal(this)) else { return false };
+    let call = make();
+    let close = matches!(&call, Call::Close);
+    let mut pending = lock(&j.pending);
+    pending.calls.push_back(call);
+    QUEUED.fetch_add(1, Relaxed);
+    if close || pending.calls.len() >= CHUNK {
+        publish(p, this, j, &mut pending);
+    }
+    let full = pending.calls.len() >= BACKLOG;
+    drop(pending);
+    if full {
+        flush(this);
+    }
+    true
+}

+impl Journal {
+    fn run(&self, this: This) {
+        loop {
+            let batch = {
+                let mut pending = lock(&self.pending);
+                if pending.calls.is_empty() {
+                    // No List access after clearing active. Release/reset
+                    // and the synchronous submitter may proceed now.
+                    pending.active = false;
+                    self.idle.notify_all();
+                    return;
+                }
+                std::mem::take(&mut pending.calls)
+            };
+            let started = Instant::now();
+            let count = batch.len() as u64;
+            for call in batch {
+                call.run(this);
+            }
+            WORK_NS.fetch_add(started.elapsed().as_nanos() as u64, Relaxed);
+            REPLAYED.fetch_add(count, Relaxed);
+            BATCHES.fetch_add(1, Relaxed);
+        }
+    }
+}
+
+/// Before any eager API call touching host/state, or any lifetime change.
+/// A worker uses only host_list_raw/with_state_now, never this wait.
+pub(crate) fn flush(this: This) {
+    let (Some(p), Some(j)) = (pool(), journal(this)) else { return };
+    let mut pending = lock(&j.pending);
+    if !pending.active && pending.calls.is_empty() {
+        return;
+    }
+    let started = Instant::now();
+    publish(p, this, j, &mut pending);
+    while pending.active || !pending.calls.is_empty() {
+        pending = j.idle.wait(pending).unwrap_or_else(|e| e.into_inner());
+        publish(p, this, j, &mut pending);
+    }
+    WAIT_NS.fetch_add(started.elapsed().as_nanos() as u64, Relaxed);
+}
+
+pub(crate) fn flush_lists(lists: &[This]) {
+    if pool().is_none() {
+        return;
+    }
+    // Publish all tails first so a small list does not serialize the others.
+    for &this in lists {
+        if let (Some(p), Some(j)) = (pool(), journal(this)) {
+            publish(p, this, j, &mut lock(&j.pending));
+        }
+    }
+    for &this in lists {
+        flush(this);
+    }
+    report();
+}
+
+fn report() {
+    static LAST: OnceLock<Mutex<Instant>> = OnceLock::new();
+    let mut last = lock(LAST.get_or_init(|| Mutex::new(Instant::now())));
+    if last.elapsed().as_secs() < 5 {
+        return;
+    }
+    let seconds = last.elapsed().as_secs_f64();
+    *last = Instant::now();
+    tracing::info!(
+        seconds,
+        queued = QUEUED.swap(0, Relaxed),
+        replayed = REPLAYED.swap(0, Relaxed),
+        batches = BATCHES.swap(0, Relaxed),
+        indirect = INDIRECT.swap(0, Relaxed),
+        worker_ms = WORK_NS.swap(0, Relaxed) as f64 / 1e6,
+        drain_ms = WAIT_NS.swap(0, Relaxed) as f64 / 1e6,
+        "COMET_THIN recording stats"
+    );
+}

+#[cfg(test)]
+mod tests {
+    use super::*;
+
+    #[test]
+    fn journal_is_fifo_and_has_only_one_ready_entry() {
+        let p = Pool::default();
+        let j = Arc::new(Journal::default());
+        let seen = Arc::new(Mutex::new(Vec::new()));
+        let this = std::ptr::null();
+        {
+            let mut pending = lock(&j.pending);
+            pending.calls.push_back(Call::Test(seen.clone(), 1));
+            publish(&p, this, &j, &mut pending);
+            pending.calls.push_back(Call::Test(seen.clone(), 2));
+            publish(&p, this, &j, &mut pending);
+        }
+        assert_eq!(lock(&p.ready).len(), 1);
+        let (_, work) = lock(&p.ready).pop_front().unwrap();
+        work.run(this);
+        assert_eq!(*lock(&seen), vec![1, 2]);
+        assert!(!lock(&j.pending).active);
+        assert!(lock(&j.pending).calls.is_empty());
+        {
+            let mut pending = lock(&j.pending);
+            pending.calls.push_back(Call::Test(seen.clone(), 3));
+            publish(&p, this, &j, &mut pending);
+        }
+        let (_, work) = lock(&p.ready).pop_front().unwrap();
+        work.run(this);
+        assert_eq!(*lock(&seen), vec![1, 2, 3]);
+    }
+
+    #[test]
+    fn pointer_arguments_are_owned_at_record_time() {
+        let mut original = vec![7, 8, 9];
+        let call = Call::Constants(true, 2, original.clone(), 1);
+        original[0] = 99;
+        assert_eq!(original[0], 99);
+        match call {
+            Call::Constants(true, 2, values, 1) => assert_eq!(values, vec![7, 8, 9]),
+            _ => panic!("wrong command"),
+        }
+    }
