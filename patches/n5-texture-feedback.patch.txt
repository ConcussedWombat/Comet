+//! GPU texture-use feedback. Observational only: never evicts or changes residency.
+use std::sync::{Arc, Mutex, Weak};
+use std::sync::atomic::{AtomicU64, Ordering::Relaxed};
+use windows::Win32::Graphics::Direct3D12::*;
+use crate::host::{HostList, host, transition};
+
+const BYTES: u64 = comet_shader::emit::TEXTURE_FEEDBACK_BITS as u64 / 8;
+const RING: usize = 3;
+static HEAPS: Mutex<Vec<Weak<Feedback>>> = Mutex::new(Vec::new());
+static NEXT: AtomicU64 = AtomicU64::new(1);
+
+struct Readback {
+    buffer: ID3D12Resource,
+    pending: Option<(u64, u64)>, // frame, submission number (0: synchronous)
+}
+struct State {
+    primed: bool,
+    ring: Vec<Readback>,
+    last_seen: Vec<u64>,
+    completed: u64,
+}
+pub(crate) struct Feedback {
+    id: u64,
+    count: u32,
+    gpu: ID3D12Resource,
+    zero: ID3D12Resource,
+    state: Mutex<State>,
+}
+
+impl Feedback {
+    pub(crate) fn new(count: u32) -> Option<Arc<Self>> {
+        let host = host()?;
+        let gpu = host.uav_buffer(BYTES)?;
+        let (zero, data) = host.upload_buffer(BYTES)?;
+        // SAFETY: fresh mapped upload of BYTES bytes; never changed afterwards.
+        unsafe { data.write_bytes(0, BYTES as usize) };
+        let mut ring = Vec::with_capacity(RING);
+        for _ in 0..RING {
+            ring.push(Readback { buffer: host.read_back_buffer(BYTES)?, pending: None });
+        }
+        let count = count.min(comet_shader::emit::TEXTURE_FEEDBACK_BITS - 1);
+        let f = Arc::new(Self {
+            id: NEXT.fetch_add(1, Relaxed), count, gpu, zero,
+            state: Mutex::new(State { primed: false, ring, last_seen: vec![0; count as usize], completed: 0 }),
+        });
+        HEAPS.lock().unwrap_or_else(|e| e.into_inner()).push(Arc::downgrade(&f));
+        tracing::info!(heap = f.id, count, bytes = BYTES, "N5 GPU texture feedback heap allocated");
+        Some(f)
+    }
+
+    pub(crate) fn bind(&self, list: &HostList) -> Option<u64> {
+        list.keep(&self.gpu);
+        // SAFETY: live GPU buffer held by list.
+        Some(unsafe { self.gpu.GetGPUVirtualAddress() })
+    }
+
+    fn collect(&self, state: &mut State, finished: u64) {
+        // Process in frame order, including delayed flights.
+        let mut ready: Vec<(u64, usize)> = state.ring.iter().enumerate()
+            .filter_map(|(i, r)| r.pending.filter(|&(_, f)| f <= finished).map(|(frame, _)| (frame, i))).collect();
+        ready.sort_unstable();
+        for (frame, i) in ready {
+            let r = &mut state.ring[i];
+            r.pending = None;
+            let Some(host) = host() else { continue; };
+            // SAFETY: corresponding copy completed; map exact buffer range.
+            unsafe {
+                if host.device.GetDeviceRemovedReason().is_err() { continue; }
+                let mut data = std::ptr::null_mut();
+                let range = D3D12_RANGE { Begin: 0, End: BYTES as usize };
+                if r.buffer.Map(0, Some(&range), Some(&mut data)).is_err() { continue; }
+                if data.is_null() { r.buffer.Unmap(0, None); continue; }
+                let words = std::slice::from_raw_parts(data.cast::<u32>(), (BYTES / 4) as usize);
+                let mut used = 0u64;
+                for (word, &bits) in words.iter().take((self.count as usize).div_ceil(32)).enumerate() {
+                    let mut bits = bits;
+                    while bits != 0 {
+                        let index = word * 32 + bits.trailing_zeros() as usize;
+                        bits &= bits - 1;
+                        if let Some(last) = state.last_seen.get_mut(index) {
+                            *last = (*last).max(frame.saturating_add(1));
+                            used += 1;
+                        }
+                    }
+                }
+                r.buffer.Unmap(0, None);
+                state.completed = state.completed.max(frame.saturating_add(1));
+                tracing::info!(heap = self.id, frame, used, slots = self.count,
+                    "N5 GPU texture feedback readback");
+                // Age estimates are diagnostic, not an eviction proof:
+                // this epoch can cover several frames and descriptor
+                // generations can change between recording and readback.
+                if frame.is_multiple_of(60) {
+                    let observed = state.last_seen.iter().filter(|&&last| last != 0).count();
+                    let cold = state.last_seen.iter().filter(|&&last| {
+                        last != 0 && state.completed.saturating_sub(last) >= 120
+                    }).count();
+                    tracing::info!(heap = self.id, frame, observed, cold,
+                        "N5 GPU texture feedback slot-age sample");
+                }
+            }
+        }
+    }
+}
+
+pub(crate) struct Ticket(Arc<Feedback>, usize, u64);
+
+/// Called on the serialized host present recording path. The device-local
+/// buffer is implicitly promoted for shader UAV access and, as a buffer,
+/// decays to COMMON at ExecuteCommandLists completion. The present list
+/// is submitted separately on that same direct queue. Its copy and reset
+/// are ordered after earlier submissions, never CPU-clearing live UAVs.
+pub(crate) fn record(list: &HostList, frame: u64, finished: u64) -> Vec<Ticket> {
+    static ON: std::sync::OnceLock<bool> = std::sync::OnceLock::new();
+    if !*ON.get_or_init(|| std::env::var_os("COMET_N5_TEXTURE_FEEDBACK").is_some_and(|v| v == "1")) {
+        return Vec::new();
+    }
+    let mut registry = HEAPS.lock().unwrap_or_else(|e| e.into_inner());
+    let live: Vec<_> = registry.iter().filter_map(Weak::upgrade).collect();
+    registry.retain(|h| h.strong_count() != 0);
+    drop(registry);
+    let mut tickets = Vec::new();
+    for f in live {
+        let mut state = f.state.lock().unwrap_or_else(|e| e.into_inner());
+        f.collect(&mut state, finished);
+        let first = !state.primed;
+        let Some(i) = state.ring.iter().position(|r| r.pending.is_none()) else {
+            // Preserve the accumulated bitmap; a later snapshot covers
+            // multiple frames. No wait and no false zero observation.
+            continue;
+        };
+        let rb = &state.ring[i].buffer;
+        list.keep(&f.gpu);
+        list.keep(&f.zero);
+        list.keep(rb);
+        // SAFETY: open present list, live buffers, exact sizes. Separate
+        // ExecuteCommandLists; buffers have decayed to COMMON.
+        unsafe {
+            if !first {
+                list.list.ResourceBarrier(&[transition(&f.gpu, D3D12_RESOURCE_STATE_COMMON, D3D12_RESOURCE_STATE_COPY_SOURCE)]);
+                list.list.CopyBufferRegion(rb, 0, &f.gpu, 0, BYTES);
+                list.list.ResourceBarrier(&[transition(&f.gpu, D3D12_RESOURCE_STATE_COPY_SOURCE, D3D12_RESOURCE_STATE_COPY_DEST)]);
+            } else {
+                list.list.ResourceBarrier(&[transition(&f.gpu, D3D12_RESOURCE_STATE_COMMON, D3D12_RESOURCE_STATE_COPY_DEST)]);
+            }
+            list.list.CopyBufferRegion(&f.gpu, 0, &f.zero, 0, BYTES);
+            list.list.ResourceBarrier(&[transition(&f.gpu, D3D12_RESOURCE_STATE_COPY_DEST, D3D12_RESOURCE_STATE_COMMON)]);
+        }
+        state.primed = true;
+        // First reset establishes known zero; never report its old bits.
+        if !first { tickets.push(Ticket(f.clone(), i, frame)); }
+    }
+    tickets
+}
+
+/// Register fence completion, not CPU frame number, as readiness.
+pub(crate) fn submitted(tickets: Vec<Ticket>, flight: u64) {
+    for Ticket(f, i, frame) in tickets {
+        f.state.lock().unwrap_or_else(|e| e.into_inner()).ring[i].pending = Some((frame, flight));
+    }
+}
