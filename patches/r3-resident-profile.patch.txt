+//! Optional cumulative timings, split between recording-time staging and
+//! submission-time resident BC refresh. No texture contents exported.
+use std::sync::OnceLock;
+use std::sync::atomic::{AtomicU64, Ordering::Relaxed};
+
+const N: usize = 8;
+const PHASES: [&str; N] = [
+    "bc_full_read", "bc_span_read", "bc_hash", "detile",
+    "layout", "footprints", "upload_alloc", "stage_zero",
+];
+static CALLS: [AtomicU64; N] = [const { AtomicU64::new(0) }; N];
+static NS: [AtomicU64; N] = [const { AtomicU64::new(0) }; N];
+static BYTES: [AtomicU64; N] = [const { AtomicU64::new(0) }; N];
+static EVENTS: AtomicU64 = AtomicU64::new(0);
+
+pub(super) fn measure<T>(phase: usize, bytes: usize, f: impl FnOnce() -> T) -> T {
+    static ON: OnceLock<bool> = OnceLock::new();
+    if !*ON.get_or_init(|| std::env::var_os("COMET_R3_RESIDENT_PROFILE").is_some_and(|v| v == "1")) {
+        return f();
+    }
+    let began = std::time::Instant::now();
+    let value = f();
+    NS[phase].fetch_add(began.elapsed().as_nanos() as u64, Relaxed);
+    CALLS[phase].fetch_add(1, Relaxed);
+    BYTES[phase].fetch_add(bytes as u64, Relaxed);
+    let event = EVENTS.fetch_add(1, Relaxed) + 1;
+    if event.is_power_of_two() || event.is_multiple_of(4096) {
+        let calls = std::array::from_fn::<_, N, _>(|i| CALLS[i].load(Relaxed));
+        let ns = std::array::from_fn::<_, N, _>(|i| NS[i].load(Relaxed));
+        let bytes = std::array::from_fn::<_, N, _>(|i| BYTES[i].load(Relaxed));
+        tracing::info!(event, phases = ?PHASES, ?calls, ?ns, ?bytes, "R3-LOAD-PROFILE");
+    }
+    value
+}
