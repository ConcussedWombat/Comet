+const HOIST_CHUNK: u64 = 256 << 10;

+fn hoist_room(bytes: u64) -> Option<u64> {
+    if bytes == 0 || bytes > HOIST_CHUNK {
+        return None;
+    }
+    Some(bytes.next_multiple_of(16))
 }

+    /// Exact requested records only; never the speculative window's tail.
+    pub(crate) fn bound(
+        &mut self,
+        src: (&ID3D12Resource, u64),
+        spec: (u32, u32, u32),
+    ) -> Option<(ID3D12Resource, u64)> {
+        let bytes = validate_layout(spec, src.1, None)?;
+        let room = hoist_room(bytes)?;
+        let c = clamp()?;
+        // SAFETY: live buffer resources held through both lists.
+        let src_base = unsafe { src.0.GetGPUVirtualAddress() };
+        let budget = unsafe { self.budget.GetGPUVirtualAddress() };
+        let (src_gpu, args_offset) = if src.1.checked_add(bytes).is_some_and(|end| end <= u64::from(u32::MAX)) {
+            (src_base, src.1 as u32)
+        } else {
+            (src_base.checked_add(src.1)?, 0)
+        };
+        if !src_gpu.is_multiple_of(4) || !budget.is_multiple_of(4) {
+            return None;
+        }
+        let l = &self.host.list;
+        if self.at + room > HOIST_CHUNK {
+            let buffer = host()?.uav_buffer(HOIST_CHUNK)?;
+            self.host.keep(&buffer);
+            // SAFETY: new scratch begins COMMON; the first global wait
+            // orders title argument producers in the preceding list.
+            unsafe {
+                if self.chunks.is_empty() {
+                    #[cfg(feature = "diagnostics")]
+                    if clamp_timing() {

+                    }
+                    l.ResourceBarrier(&[uav_barrier()]);
+                    #[cfg(feature = "diagnostics")]
+                    crate::copystat::add(crate::copystat::C::IndirectGlobalWaits, 1);
+                }
+                l.ResourceBarrier(&[transition(&buffer, D3D12_RESOURCE_STATE_COMMON,
+                    D3D12_RESOURCE_STATE_UNORDERED_ACCESS)]);
+            }
+            self.chunks.push(buffer);
+            self.at = 0;
+        }
+        let dst = self.chunks.last()?.clone();
+        let at = self.at;
+        self.at += room;

+        let constants = [kind, stride, records, args_offset, 0, 0];
+        // SAFETY: live scratch, source and budget; disjoint 16-byte-aligned
+        // output spans. A resource-specific UAV barrier orders budget charges
+        // exactly between calls, without draining unrelated title UAVs.
+        #[cfg(feature = "diagnostics")]
+        self.host.note_op(|| format!("B4 hoisted clamp kind {kind} max {records}"));

+            l.SetPipelineState(&c.pso);
+            l.SetComputeRoot32BitConstants(0, 6, constants.as_ptr().cast(), 0);
+            l.SetComputeRootShaderResourceView(1, src_gpu);
+            l.SetComputeRootShaderResourceView(2, src_gpu);

+            l.SetComputeRootUnorderedAccessView(4, budget);

+            let mut barrier = uav_barrier();

+                pResource: std::mem::transmute_copy(&self.budget),

+        }

+    }

+    /// Publish every chunk before any consumer; retain it on the consumer,
+    /// and return it to COMMON after the last draw (including repeated runs).
+    pub(crate) fn finish(self, draws: &HostList) -> HostList {
+        for buffer in &self.chunks {
+            draws.keep(buffer);
+        }
+        let publish: Vec<_> = self.chunks.iter().map(|buffer| transition(buffer,
+            D3D12_RESOURCE_STATE_UNORDERED_ACCESS, D3D12_RESOURCE_STATE_INDIRECT_ARGUMENT)).collect();
+        let release: Vec<_> = self.chunks.iter().map(|buffer| transition(buffer,
+            D3D12_RESOURCE_STATE_INDIRECT_ARGUMENT, D3D12_RESOURCE_STATE_COMMON)).collect();
+        // SAFETY: both lists open; their ordered execution separates these barriers.
+        unsafe {
+            if !publish.is_empty() {
+                #[cfg(feature = "diagnostics")]
+                if clamp_timing() {
+                    self.host.note_op(|| "B4 hoist-publish".to_string());
+                }
+                self.host.list.ResourceBarrier(&publish);
+                draws.list.ResourceBarrier(&release);
+            }
+        }
+        if self.records != 0 {
+            static RUNS: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
+            let n = RUNS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
+            if n < 16 || n.is_multiple_of(512) {
+                tracing::info!(calls = self.records, chunks = self.chunks.len(), "B4 hoisted indirect clamp run");
+            }
+        }
+        self.host.close();
+        self.host
+    }
