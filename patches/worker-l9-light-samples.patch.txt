+fn l9_sample_coordinates() -> &'static [(u32, u32, u32, u32)] {
+    static POINTS: std::sync::OnceLock<Vec<(u32, u32, u32, u32)>> = std::sync::OnceLock::new();
+    POINTS.get_or_init(|| {
+        std::env::var("COMET_L9_LIGHT_SAMPLES").unwrap_or_default()
+            .split(',').filter_map(|s| {
+                let (size, xy) = s.trim().split_once(':')?;

+                let (x, y) = xy.split_once(':')?;

+                (x < w && y < h).then_some((w, h, x, y))

+    })

+/// Raw bytes only, based on the *copy footprint* format. Unknown formats
+/// are intentionally skipped; no float/UNORM/sRGB or packed-light inference.
+fn l9_texel_bytes(format: i32, plane: u32) -> Option<usize> {
+    if plane == 1 {
+        // Same one-byte stencil plane layout used by stencil_values.
+        return matches!(format, 60..=65).then_some(1);
+    }
+    match format {
+        1..=4 => Some(16),             // RGBA32
+        5..=8 => Some(12),             // RGB32
+        9..=18 => Some(8),             // RGBA16 / RG32
+        23..=47 | 67 | 87..=93 => Some(4),
+        48..=59 => Some(2),
+        60..=65 => Some(1),
+        _ => None,
+    }
+}

+fn l9_texel_at(rows: &[u8], pitch: usize, x: usize, y: usize, bytes: usize) -> Option<&[u8]> {
+    let col = x.checked_mul(bytes)?;
+    let end_col = col.checked_add(bytes)?;
+    if end_col > pitch { return None; }
+    let row = y.checked_mul(pitch)?;
+    rows.get(row.checked_add(col)?..row.checked_add(end_col)?)
+}

+/// Classify only explicitly floating footprints, never reinterpret UINT.
+fn l9_float_classes(format: i32, raw: &[u8]) -> Option<Vec<&'static str>> {
+    let class = |bits: u32, exp: u32, mant: u32| {
+        let exponent = (bits >> mant) & ((1 << exp) - 1);
+        let fraction = bits & ((1 << mant) - 1);
+        match (exponent, fraction) {
+            (0, 0) => "zero",
+            (0, _) => "subnormal",
+            (e, 0) if e == (1 << exp) - 1 => "infinity",
+            (e, _) if e == (1 << exp) - 1 => "nan",
+            _ => "finite",
+        }
+    };
+    match format {
+        2 | 16 | 41 => Some(raw.chunks_exact(4).map(|b|
+            class(u32::from_le_bytes(b.try_into().unwrap()), 8, 23)).collect()),
+        10 => Some(raw.chunks_exact(2).map(|b|
+            class(u32::from(u16::from_le_bytes(b.try_into().unwrap())), 5, 10)).collect()),
+        26 => {
+            let b = u32::from_le_bytes(raw.try_into().ok()?);
+            Some(vec![class(b, 5, 6), class(b >> 11, 5, 6), class(b >> 22, 5, 5)])
+        }
+        _ => None,
+    }
+}

+    plane: u32, f: &D3D12_SUBRESOURCE_FOOTPRINT, rows: &[u8],

+    let points = l9_sample_coordinates();
+    if points.is_empty() { return; }

+                plane, w, h, x, y, ?raw, ?classes, "L9-LIGHT-SAMPLE end-of-frame raw texel");
+        }
     }
