+/// Explicit native-target coordinates, not scaled PNG coordinates. Only
+/// run on existing capture readbacks; no extra GPU work or format guessing.
+fn l9_sample_coordinates() -> &'static [(u32, u32, u32, u32)] {
+    static POINTS: std::sync::OnceLock<Vec<(u32, u32, u32, u32)>> = std::sync::OnceLock::new();
+    POINTS.get_or_init(|| {
+        std::env::var("COMET_L9_LIGHT_SAMPLES").unwrap_or_default()
+            .split(',').filter_map(|s| {
+                let (size, xy) = s.trim().split_once(':')?;
+                let (w, h) = size.split_once('x')?;
+                let (x, y) = xy.split_once(':')?;
+                let (w, h, x, y): (u32, u32, u32, u32) =
+                    (w.parse().ok()?, h.parse().ok()?, x.parse().ok()?, y.parse().ok()?);
+                (x < w && y < h).then_some((w, h, x, y))
+            }).take(16).collect()
+    })
+}

+/// Raw bytes only, based on the *copy footprint* format. Unknown formats
+/// are intentionally skipped; no float/UNORM/sRGB or packed-light inference.
+fn l9_texel_bytes(format: i32, plane: u32) -> Option<usize> {
+    if plane == 1 {
+        // Same one-byte stencil plane layout used by stencil_values.
+        return Some(1);
+    }
+    match format {
+        1..=4 => Some(16),             // RGBA32
+        9..=18 => Some(8),   // RGBA16 / RG32
+        23..=47 | 67 | 87..=93 => Some(4),
+        48..=59 => Some(2),
+        60..=65 => Some(1),
+        _ => None,
+    }
+}

+fn l9_texel_at(rows: &[u8], pitch: usize, x: usize, y: usize, bytes: usize) -> Option<&[u8]> {
+    let col = x.checked_mul(bytes)?;
+    let end_col = col.checked_add(bytes)?;
+    if end_col > pitch { return None; }
+    let row = y.checked_mul(pitch)?;
+    rows.get(row.checked_add(col)?..row.checked_add(end_col)?)
+}
