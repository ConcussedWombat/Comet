+/// Bounded sample via the checked title-memory reader; missing pages are not a valid fingerprint.
+pub(crate) fn n5_fingerprint(at: u64, len: u64) -> Option<u64> {
+    const WINDOW: usize = 16 << 10;
+    let len = usize::try_from(len).ok()?;
+    let base = usize::try_from(at).ok()?;
+    if len == 0 || base.checked_add(len).is_none() { return None; }
+    let mut hash = blake3::Hasher::new();
+    with_read(len.min(3 * WINDOW), |read| {
+        let mut take = |offset: usize, bytes: usize| {
+            if !crate::host::read_into((base + offset) as *const u8, &mut read[..bytes]) { return false; }
+            hash.update(&read[..bytes]);
+            true
+        };
+        if len <= 3 * WINDOW { take(0, len) } else {
+            take(0, WINDOW) && take((len / 2) & !0xfff, WINDOW) && take(len - WINDOW, WINDOW)
+        }
+    }).then(|| u64::from_le_bytes(hash.finalize().as_bytes()[..8].try_into().expect("eight bytes")))
+}
