+pub(crate) fn w6_present_age_trace() -> bool {
+    static ON: OnceLock<bool> = OnceLock::new();
+    *ON.get_or_init(|| std::env::var_os("COMET_W6_PRESENT_AGE_TRACE").is_some_and(|v| v == "1"))

+fn w6_present_age_enabled() -> bool {
+    w6_hold_stale_present() || w6_present_age_trace()
+}
+
+// Use weak identity: a released texture can never match a new allocation.
+// The history keeps no GPU texture alive, and is bounded to eight sources.
+type W6Seen<T> = Vec<(std::sync::Weak<T>, u64)>;
+
+fn w6_seen_stale<T>(seen: &W6Seen<T>, source: &std::sync::Arc<T>, serial: u64) -> bool {
+    let key = std::sync::Arc::downgrade(source);
+    serial != 0
+        && seen.last().is_some_and(|(last, latest)| !last.ptr_eq(&key) && *latest > serial)
+        && seen.iter().any(|(old, used)| old.ptr_eq(&key) && *used == serial)
+}
+
+fn w6_remember_present<T>(seen: &mut W6Seen<T>, source: &std::sync::Arc<T>, serial: u64) {
+    let key = std::sync::Arc::downgrade(source);
+    seen.retain(|(old, _)| !old.ptr_eq(&key));
+    if seen.len() >= 8 {
+        seen.remove(0);
+    }
+    seen.push((key, serial));
+}
+
+#[cfg(test)]
+#[test]
+fn w6_present_age_only_holds_known_older_unchanged_sources() {
+    let a = std::sync::Arc::new(1);
+    let b = std::sync::Arc::new(2);
+    let c = std::sync::Arc::new(3);
+    let mut seen = Vec::new();
+    assert!(!w6_seen_stale(&seen, &a, 1));
+    w6_remember_present(&mut seen, &a, 1);
+    w6_remember_present(&mut seen, &b, 2);
+    w6_remember_present(&mut seen, &c, 3);
+    assert!(w6_seen_stale(&seen, &a, 1));
+    assert!(w6_seen_stale(&seen, &b, 2));
+    assert!(!w6_seen_stale(&seen, &c, 3));
+    assert!(!w6_seen_stale(&seen, &a, 4));
+    assert!(!w6_seen_stale(&seen, &a, 0));
+    w6_remember_present(&mut seen, &a, 4);
+    assert!(w6_seen_stale(&seen, &c, 3));
+    assert!(!w6_seen_stale(&seen, &a, 4));
+    drop(b);
+    let replacement = std::sync::Arc::new(2);
+    assert!(!w6_seen_stale(&seen, &replacement, 2));
+}
