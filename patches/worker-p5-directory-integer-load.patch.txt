+/// Raw directory entries are two u32 words, not floating-point values.
+/// Same raw SRV, address, mask and alignment in both forms; the integer
+/// overload avoids a bitcast per extracted component. This is not a
+/// residency shortcut: no lookup or fault reporting is removed.
+fn p5_directory_words(s: &mut Shader, c: usize, args: &[V], integer: bool) -> (V, V) {
+    let i32t = s.t.i32;
+    let (name, ret) = if integer {
+        ("rawBufferLoad.i32", s.m.named_struct("dx.types.ResRet.i32", vec![i32t; 5]))
+    } else {
+        ("rawBufferLoad.f32", s.t.res_ret)
+    };
+    let r = s.op(c, name, Attr::ReadOnly, ret, 139, args);
+    let (copy, start) = (s.m.extract(c, r, 0), s.m.extract(c, r, 1));
+    if integer {
+        (copy, start)
+    } else {
+        (
+            s.m.cast(c, Cast::Bitcast, copy, i32t),
+            s.m.cast(c, Cast::Bitcast, start, i32t),
+        )
+    }
+}
