+/// Belobug 2aad398 doubled the dynamic reserve. Keep Comet's stride
+/// directory large enough for every possible descriptor index, and do not
+/// take additional mirrored title slots away on a capped host.
+fn b2_dynamic_reserve(count: u32, enabled: bool) -> u32 {
+    if enabled && count <= comet_shader::emit::SLOT_STRIDES_COUNT - 2 * DYNAMIC {
+        2 * DYNAMIC
+    } else {
+        DYNAMIC
+    }
+}
