+fn l6_interp_sites(p: &NativeProgram, program: &ir::Program, raised: Option<&io::Io>, label: &str) {
+    if p.stage != HwStage::Ps
+        || !std::env::var_os("COMET_L6_INTERP_SITES").is_some_and(|v| v == "1")
+    {
+        return;
+    }
+    use comet_shader::gfx10::{Fields, Operand};
+
+    let mut defs = BTreeMap::new();
+    let mut joins = BTreeMap::new();
+    let mut reads = BTreeMap::<ir::Value, usize>::new();
+    for (b, block) in program.blocks.iter().enumerate() {
+        for phi in &block.phis {
+            joins.insert(phi.value, b);
+            for value in &phi.args {
+                *reads.entry(*value).or_default() += 1;
+            }
+        }
+        for inst in &block.insts {
+            for &(_, value) in &inst.writes {
+                defs.insert(value, (inst.machine.offset, inst.name()));
+            }
+            for &(_, value) in &inst.reads {
+                *reads.entry(value).or_default() += 1;
+            }
+        }
+    }
+    let origin = |value: Option<ir::Value>| match value {
+        Some(v) => match defs.get(&v) {
+            Some((offset, name)) => format!("{name}@{offset:#x}"),
+            None => match joins.get(&v) {
+                Some(b) => format!("phi@block{b}"),
+                None => "entry".to_owned(),
+            },
+        },
+        None => "not-vgpr".to_owned(),
+    };
+    let read = |inst: &ir::Inst, op: Option<&Operand>| match op {
+        Some(Operand::Vector { reg, .. }) => inst.read(ir::Reg::V(*reg)),
+        _ => None,
+    };
+    let sites: BTreeMap<_, _> = raised.into_iter()
+        .flat_map(|io| io.interps.iter().map(|(s, interp)| (*s, interp)))
+        .collect();
+    for (b, block) in program.blocks.iter().enumerate() {
+        for (i, inst) in block.insts.iter().enumerate() {
+            let name = inst.name();
+            if !name.starts_with("v_interp_") {
+                continue;
+            }
+            let (dst, weight, acc, opsel) = match &inst.fields {
+                Fields::Vintrp { dst, src, .. } => (
+                    Some(dst), Some(src), name.starts_with("v_interp_p2").then_some(dst), 0,
+                ),

+                ),

+            };

+            let acc = origin(read(inst, acc));
