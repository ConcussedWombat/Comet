+        };

+        let i = self.x.p.blocks[b].insts.iter().find(|i| i.writes.first().is_some_and(|w| w.1 == value))?;

+        let Fields::Sop { src, .. } = &i.fields else { return None; };

+/// Draw-evaluable descriptor address encoded in RootType::Other.
+/// UserData::dword is its constant BYTE displacement.
+#[derive(Clone, Copy, Debug, PartialEq, Eq)]
+pub struct L5SrdIndex {
+    pub heap: bool,
+    pub low30: bool,
+    pub constant: Option<(u8, u8, u8)>, // slot, dword, shift
+}
+impl L5SrdIndex {
+    const PREFIX: u32 = 0x5100_0000;
+    pub fn encode(self) -> u32 {
+        let mut kind = Self::PREFIX | (u32::from(self.heap) << 23)
+            | (u32::from(self.low30) << 22);
+        if let Some((slot, dword, shift)) = self.constant {
+            kind |= (1 << 21) | (u32::from(shift & 31) << 16)
+                | (u32::from(dword) << 8) | u32::from(slot);
+        }
+        kind
+    }
+    pub fn decode(kind: u32) -> Option<Self> {
+        if kind & 0xff00_0000 != Self::PREFIX { return None; }
+        Some(Self {
+            heap: kind & (1 << 23) != 0,
+            low30: kind & (1 << 22) != 0,
+            constant: (kind & (1 << 21) != 0)
+                .then_some((kind as u8, (kind >> 8) as u8, ((kind >> 16) & 31) as u8)),
+        })
+    }
+    pub fn byte_offset(self, base: u32, offset: u32, constant: u32) -> Option<u32> {
+        base.checked_add(offset)?.checked_add(self.constant
+            .map_or(0, |(_, _, shift)| constant.wrapping_shl(u32::from(shift))))
+    }
