+fn p1_twin_pair(b: &Barrier, colour: This, plane: This) -> Option<[Barrier; 2]> {
+    // Only the known legacy named UAV barrier, never transitions, aliases,
+    // split/unknown flags or the title's explicit global UAV barrier.
+    if b.kind != 2 || b.flags != 0 || b.first.is_null()
+        || colour.is_null() || plane.is_null() || colour == plane
+    {
+        return None;
+    }
+    Some([
+        Barrier { first: colour, ..UAV_BARRIER },
+        Barrier { first: plane, ..UAV_BARRIER },
+    ])
+}

+#[cfg(windows)]
+fn p1_twin_barriers(list: &crate::host::HostList, b: &Barrier) -> Option<[Barrier; 2]> {
+    use windows::core::Interface;
+    if b.kind != 2 || b.flags != 0 || b.first.is_null() {
+        return None;
+    }
+    // One Arc identifies the exact pair: do not independently look up
+    // the two host resources, or try to infer aliases from title addresses.
+    let twin = crate::resource::twin(b.first)?;
+    let plane = twin.plane.as_ref()?;
+    let colour = &twin.colour.texture;
+    let plane = &plane.texture;
+    let pair = p1_twin_pair(

+        colour.as_raw().cast_const().cast(),
+        plane.as_raw().cast_const().cast(),
+    )?;
+    // Barrier pointers borrow these COM objects. Retain both through list
+    // reset even if the title releases the resource before submission.
+    list.keep(colour);
+    list.keep(plane);
+    #[cfg(feature = "diagnostics")]
+    crate::copystat::add(crate::copystat::C::P1TwinWaits, 1);
+    Some(pair)
