+// The successful ahead ranges are unchanged. Only their answer timestamp
+// advances: each shadow still has to consume the per-page write at least
+// once before it can be current. A partial refresh remains stale.
+fn record_submission_write(j: &mut Journal, at: usize, len: usize, advance: bool) {
+    j.asked += 1;
+    let now = j.asked;
+    if advance {
+        j.ahead_asked = now;
+    }
+    for page in at / PAGE..at.saturating_add(len).div_ceil(PAGE) {
+        j.written.insert(page, now);
+    }
+}

+#[cfg(test)]
+#[test]
+fn known_submission_write_advances_only_the_standing_epoch() {
+    for advance in [false, true] {
+        let mut j = Journal {
+            asked: 10,
+            written: BTreeMap::from([(1, 8), (3, 9)]),
+            found: Vec::new(),
+            ahead: vec![(1, 4)],
+            ahead_asked: 10,
+            kept: true,
+        };
+        record_submission_write(&mut j, 2 * PAGE + 7, 1, advance);
+        assert_eq!(j.asked, 11);
+        assert_eq!(j.written, BTreeMap::from([(1, 8), (2, 11), (3, 9)]));
+        assert_eq!(j.ahead, vec![(1, 4)]);
+        assert!(j.kept);
+        // Before a full-page refresh a copy stamped at 10 is stale.
+        assert!(j.written[&2] > 10);
+        // copy_stale stamps a full refresh with the standing answer's
+        // epoch. Only the enabled path stops asking for the same write.
+        assert_eq!(j.written[&2] > j.ahead_asked, !advance);
+        record_submission_write(&mut j, 2 * PAGE + 7, 1, advance);
+        assert_eq!(j.written[&2], 12);
+        assert_eq!(j.ahead_asked, if advance { 12 } else { 10 });
+        assert_eq!(j.written[&1], 8);
+        assert_eq!(j.written[&3], 9);
+    }
+}
