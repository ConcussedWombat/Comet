+/// Exact-range host copies for otherwise unrepresentable buffer views.
+type ViewCopies = BTreeMap<(u64, u64), Arc<Shadow>>;
+static VIEW_COPIES: Mutex<ViewCopies> = Mutex::new(BTreeMap::new());
+
+/// Keyed by title byte range so view creation, refresh and copy-back
+/// agree. Do not replace a copy still held by a slot or a submitted list.
+/// Cap this experimental cache; a failed allocation leaves the view null.
+pub(crate) fn view_bytes(start: u64, len: u64) -> Option<Arc<Shadow>> {
+    const MAX_COPIES: usize = 1024;
+    const MAX_BYTES: u64 = 256 << 20;
+    start.checked_add(len)?;
+    let size = len.checked_add(255)? & !255;
+    if len == 0 || size > MAX_BYTES {
+        return None;
+    }
+    let mut copies = VIEW_COPIES.lock().unwrap_or_else(|e| e.into_inner());
+    if let Some(copy) = copies.get(&(start, len)) {
+        return Some(copy.clone());
+    }
+    let resident: u64 = copies.keys().map(|&(_, n)| (n + 255) & !255).sum();
+    if copies.len() >= MAX_COPIES || resident + size > MAX_BYTES {
+        copies.retain(|_, copy| {
+            Arc::strong_count(copy) > 1 || crate::host::references(&copy.buffer) > 1
+        });
+    }
+    let resident: u64 = copies.keys().map(|&(_, n)| (n + 255) & !255).sum();
+    if copies.len() >= MAX_COPIES || resident + size > MAX_BYTES {
+        once!("N3 own-view copy cache full; view left null");
+        return None;
+    }
+    let (buffer, data) = host()?.shared_buffer(size)?;
+    crate::host::name(&buffer, || format!("title own-view {start:#x}+{len:#x}"));
+    // SAFETY: a live buffer.
+    let gpu = unsafe { buffer.GetGPUVirtualAddress() };
+    let copy = Arc::new(Shadow::new(buffer, data, gpu, start, len));
+    copies.insert((start, len), copy.clone());
+    tracing::info!(start, len, size, "N3 own-view shadow allocated");
+    Some(copy)
