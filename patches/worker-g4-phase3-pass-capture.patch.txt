+/// The title pipeline identity, even when no host PSO exists. No wait,
+/// queueing or translation is triggered by the capture.
+#[cfg(all(windows, feature = "diagnostics"))]
+pub(crate) fn g4_pass_identity(this: This, recorded: bool) -> (String, bool, String) {
+    if this.is_null() {
+        return ("none".into(), false, "no pipeline set".into());
+    }
+    let p = pipeline(this);
+    let selected = p.shaders.iter().any(|s| matches!(s.hash.as_str(),
+        "2da4beb386509b77" | "369cbe906ea72764" | "2cc4c197ee29e291" |
+        "02384a60fda6f2d0" | "6b82bc7d22016cb5" |
+        "f5e50eb2108c425b" | "ce3f0432a08f1d35" |
+        "c89489afd89ee7f0" | "45f081396c2c59ba" | "63083c777948927a" |
+        "30332cf87c4e0eae" | "8563c39d693b36d0" | "ba8d52f4ea1a1bac" |
+        "9146291b38ea9ce1" | "42683eebec081520"));
+    let hashes = p.shaders.iter().map(|s| format!("{}:{}", s.stage, s.hash))
+        .collect::<Vec<_>>().join(",");
+    let status = if recorded {
+        "recorded".into()
+    } else {
+        skipped_shader(p).map(|h| format!("skipped shader {h}"))
+            .or_else(|| make::missing(&p.host))
+            .unwrap_or_else(|| "host PSO ready at observation".into())
+    };
+    (hashes, selected, status)
+}

+                    }
+                }).collect::<Vec<_>>().join(",")).unwrap_or_default()
+            } else {
+                String::new()
+            };
+            format!("G4-PASS kind={kind} recorded={recorded} pipeline={:#x} shaders=[{hashes}] targets={targets} status={status:?} selected={selected} roots=[{roots}] {}", self.pipeline, detail())
+        });
+    }
