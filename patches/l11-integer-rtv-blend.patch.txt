+// Guest integer color targets bypass blending. Keep the float targets and
+// every active write mask intact when D3D12 requires per-target state.
+fn l11_integer_rtv(format: DXGI_FORMAT) -> bool {
+    matches!(
+        format,
+        DXGI_FORMAT_R32G32B32A32_UINT | DXGI_FORMAT_R32G32B32A32_SINT
+            | DXGI_FORMAT_R32G32B32_UINT | DXGI_FORMAT_R32G32B32_SINT
+            | DXGI_FORMAT_R16G16B16A16_UINT | DXGI_FORMAT_R16G16B16A16_SINT
+            | DXGI_FORMAT_R32G32_UINT | DXGI_FORMAT_R32G32_SINT
+            | DXGI_FORMAT_R10G10B10A2_UINT
+            | DXGI_FORMAT_R8G8B8A8_UINT | DXGI_FORMAT_R8G8B8A8_SINT
+            | DXGI_FORMAT_R16G16_UINT | DXGI_FORMAT_R16G16_SINT
+            | DXGI_FORMAT_R32_UINT | DXGI_FORMAT_R32_SINT
+            | DXGI_FORMAT_R8G8_UINT | DXGI_FORMAT_R8G8_SINT
+            | DXGI_FORMAT_R16_UINT | DXGI_FORMAT_R16_SINT
+            | DXGI_FORMAT_R8_UINT | DXGI_FORMAT_R8_SINT
+    )
+}

+fn l11_integer_blend(
+    mut blend: D3D12_BLEND_DESC,
+    targets: &D3D12_RT_FORMAT_ARRAY,
+) -> Option<D3D12_BLEND_DESC> {
+    let formats = &targets.RTFormats[..targets.NumRenderTargets.min(8) as usize];
+    let shared = !blend.IndependentBlendEnable.as_bool();
+    if !formats.iter().enumerate().any(|(i, &format)| {
+        l11_integer_rtv(format) && blend.RenderTarget[if shared { 0 } else { i }].BlendEnable.as_bool()
+    }) {
+        return None;
+    }
+    if shared {
+        let common = blend.RenderTarget[0];
+        blend.RenderTarget[..formats.len()].fill(common);
+        blend.IndependentBlendEnable = (formats.len() > 1).into();
+    }
+    for (format, target) in formats.iter().zip(&mut blend.RenderTarget) {
+        if l11_integer_rtv(*format) {
+            target.BlendEnable = false.into();
+        }
+    }
+    Some(blend)
+}

+fn l11_integer_blend_enabled() -> bool {
+    static ENABLED: std::sync::OnceLock<bool> = std::sync::OnceLock::new();
+    *ENABLED.get_or_init(|| std::env::var_os("COMET_L11_INTEGER_BLEND").is_some_and(|v| v == "1"))
+}
