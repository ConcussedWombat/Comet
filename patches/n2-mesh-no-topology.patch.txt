+/// The final stream member is the topology subobject. Its offset is the
+/// exact byte size of the preceding complete pointer-aligned subobjects.
+/// Avoid the pipeline library for this experimental variant: see 972/970
+/// above. No alternate cached PSO or modified shader is substituted.
+fn n2_mesh_direct<S>(
+    device: &ID3D12Device,
+    stream: &mut S,
+    prefix_size: usize,
+) -> windows::core::Result<ID3D12PipelineState> {
+    use windows::core::Interface;
+    let desc = D3D12_PIPELINE_STATE_STREAM_DESC {
+        SizeInBytes: prefix_size,
+        pPipelineStateSubobjectStream: std::ptr::from_mut(stream).cast(),
+    };
+    let device = device.cast::<ID3D12Device2>()?;
+    // SAFETY: caller passes a complete prefix of the live stream.
+    unsafe { device.CreatePipelineState(&desc) }
