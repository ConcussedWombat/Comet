+#[derive(Debug, Clone, Copy)]
+struct L3Channel {
+    finite: u64,
+    zero: u64,
+    negative: u64,
+    nan: u64,
+    infinite: u64,
+    min: f64,
+    max: f64,
+    sum: f64,
+}
+
+impl Default for L3Channel {
+    fn default() -> Self {
+        Self {
+            finite: 0, zero: 0, negative: 0, nan: 0, infinite: 0,
+            min: f64::INFINITY, max: f64::NEG_INFINITY, sum: 0.0,
+        }
+    }
+}
+
+impl L3Channel {
+    fn add(&mut self, v: f64) {
+        if v.is_nan() {
+            self.nan += 1;
+        } else if v.is_infinite() {
+            self.infinite += 1;
+        } else {
+            self.finite += 1;
+            self.zero += u64::from(v == 0.0);
+            self.negative += u64::from(v < 0.0);
+            self.min = self.min.min(v);
+            self.max = self.max.max(v);
+            self.sum += v;
+        }
+    }
+}
+
+/// Existing visualization helpers collapse NaNs to infinity. Keep the
+/// nonfinite classes distinct for the numeric diagnostic.
+fn l3_half(v: u16) -> f64 {
+    if v & 0x7c00 == 0x7c00 && v & 0x03ff != 0 {
+        f64::NAN
+    } else {
+        f64::from(half(v))
+    }
+}
+
+fn l3_ufloat(v: u32, mantissa: u32) -> f64 {
+    let mask = (1 << mantissa) - 1;
+    if v >> mantissa == 31 && v & mask != 0 {
+        f64::NAN
+    } else {
+        f64::from(small_float(v, mantissa))
+    }
+}

+#[derive(Debug)]
+struct L3LightStats {
+    interpretation: &'static str,
+    count: usize,
+    color_channels: usize,
+    pixels: u64,
+    color_zero: u64,
+    channels: [L3Channel; 4],
+}

+/// Only decode explicitly typed footprints. In particular R32_UINT is
+/// not necessarily RGB9E5, and typeless storage is not proof of a view
+/// format. For sRGB the reported values are encoded, not linear light.
+fn l3_light_stats(
+    format: u32, w: usize, h: usize, pitch: usize, rows: &[u8],
+) -> Option<L3LightStats> {
+    let (size, count, color_channels, interpretation) = match format {
+        2 => (16, 4, 3, "float32"),
+        10 => (8, 4, 3, "float16"),
+        16 => (8, 2, 2, "float32"),
+        34 => (4, 2, 2, "float16"),
+        41 => (4, 1, 1, "float32"),
+        54 => (2, 1, 1, "float16"),
+        26 => (4, 3, 3, "r11g11b10_float"),
+        67 => (4, 3, 3, "rgb9e5_sharedexp"),
+        42 => (4, 1, 1, "uint32_raw_not_light"),
+        28 | 87 => (4, 4, 3, "unorm8"),
+        29 | 91 => (4, 4, 3, "srgb8_encoded"),
+        61 => (1, 1, 1, "unorm8"),
+        _ => return None,
+    };
+    let row_bytes = w.checked_mul(size)?;
+    if w == 0 || h == 0 || row_bytes > pitch
+        || (h - 1).checked_mul(pitch)?.checked_add(row_bytes)? > rows.len()
+    {
+        return None;
+    }
+    let mut stats = L3LightStats {
+        interpretation, count, color_channels, pixels: 0, color_zero: 0,
+        channels: [L3Channel::default(); 4],
+    };
+    for y in 0..h {
+        let row = &rows[y * pitch..y * pitch + row_bytes];
+        for b in row.chunks_exact(size) {
+            let word = |i: usize| u32::from_le_bytes(
+                b[i * 4..i * 4 + 4].try_into().expect("dword"));
+            let halfword = |i: usize| u16::from_le_bytes(
+                b[i * 2..i * 2 + 2].try_into().expect("halfword"));
+            let mut v = [0.0; 4];
+            match format {
+                2 | 16 | 41 => {
+                    for (i, c) in v[..count].iter_mut().enumerate() {
+                        *c = f64::from(f32::from_bits(word(i)));
+                    }
+                }
+                10 | 34 | 54 => {
+                    for (i, c) in v[..count].iter_mut().enumerate() {
+                        *c = l3_half(halfword(i));
+                    }
+                }
+                26 => {
+                    let p = word(0);
+                    v[..3].copy_from_slice(&[
+                        l3_ufloat(p & 0x7ff, 6),
+                        l3_ufloat(p >> 11 & 0x7ff, 6),
+                        l3_ufloat(p >> 22, 5),
+                    ]);
+                }
+                67 => {
+                    let p = word(0);
+                    let scale = 2f64.powi((p >> 27) as i32 - 24);
+                    for (i, c) in v[..3].iter_mut().enumerate() {
+                        *c = f64::from(p >> (9 * i) & 0x1ff) * scale;
+                    }
+                }
+                42 => v[0] = f64::from(word(0)),
+                28 | 29 | 87 | 91 | 61 => {
+                    for (i, c) in v[..count].iter_mut().enumerate() {
+                        *c = f64::from(b[i]) / 255.0;
+                    }
+                    if matches!(format, 87 | 91) {
+                        v.swap(0, 2);
+                    }
+                }
+                _ => unreachable!(),
+            }
+            stats.pixels += 1;
+            stats.color_zero += u64::from(v[..color_channels].iter().all(|&c| c == 0.0));
+            for (c, val) in stats.channels[..count].iter_mut().zip(v) {
+                c.add(val);
+            }
+        }
+    }
+    Some(stats)
+}

+fn l3_report_light(
+    frame: u64, index: usize, resource: crate::object::This,
+    resource_format: i32, f: &D3D12_SUBRESOURCE_FOOTPRINT, rows: &[u8],
+) {
+    let Some(s) = l3_light_stats(
+        f.Format.0 as u32, f.Width as usize, f.Height as usize, f.RowPitch as usize, rows,
+    ) else {
+        tracing::info!(frame, index, ?resource, resource_format,
+            footprint_format = f.Format.0, "L3 LIGHT unsupported typed footprint");
+        return;
+    };
+    let channels = &s.channels[..s.count];
+    let mean: Vec<Option<f64>> = channels.iter()
+        .map(|c| (c.finite != 0).then(|| c.sum / c.finite as f64))
+        .collect();
+    tracing::info!(frame, index, ?resource, resource_format, footprint_format = f.Format.0,
+        subresource = 0, width = f.Width, height = f.Height,
+        interpretation = s.interpretation, pixels = s.pixels,
+        color_channels = s.color_channels, color_zero = s.color_zero,
+        ?channels, ?mean, "L3 LIGHT target end-of-frame");
+}
