+/// An exact slice of a root constant, scaled into a byte offset. The
+/// selected bits are contiguous; the result uses modulo-2^32 SALU math.
+/// This is a proof domain, not an approximation of arbitrary indices.
+#[derive(Clone, Copy, Debug, PartialEq, Eq)]
+pub struct L7RootBits {
+    pub slot: u8,
+    pub word: u8,
+    pub start: u8,
+    pub width: u8,
+    pub shift: u8,
+}
+
+impl L7RootBits {
+    pub fn root(slot: u8, word: u8) -> Option<Self> {
+        // Fits the private host-word encoding below. Larger root signatures
+        // are not silently truncated.
+        (slot < 64 && word < 32).then_some(Self { slot, word, start: 0, width: 32, shift: 0 })
+    }
+
+    fn low_mask(width: u8) -> u32 {
+        u32::MAX >> (32 - u32::from(width))
+    }
+
+    pub fn left(mut self, amount: u32) -> Option<Self> {
+        let shift = u32::from(self.shift) + (amount & 31);
+        if shift >= 32 { return None; }
+        self.shift = shift as u8;
+        self.width = self.width.min(32 - self.shift);
+        Some(self)
+    }
+
+    pub fn right(mut self, amount: u32) -> Option<Self> {
+        let amount = (amount & 31) as u8;
+        if amount <= self.shift {
+            self.shift -= amount;
+        } else {
+            let discarded = amount - self.shift;
+            if discarded >= self.width { return None; }
+            self.shift = 0;
+            self.start += discarded;
+            self.width -= discarded;
+        }
+        Some(self)
+    }
+
+    pub fn and(mut self, mask: u32) -> Option<Self> {
+        // Intersect with the live output field. Only a contiguous subset
+        // of those bits remains representable.
+        let relevant = (mask >> self.shift) & Self::low_mask(self.width);
+        if relevant == 0 { return None; }
+        let first = relevant.trailing_zeros() as u8;
+        let width = (32 - relevant.leading_zeros()) as u8 - first;
+        if relevant >> first != Self::low_mask(width) { return None; }
+        self.start += first;
+        self.shift += first;
+        self.width = width;
+        Some(self)
+    }
+
+    pub fn extract(self, offset: u32, width: u32) -> Option<Self> {
+        let offset = offset & 31;
+        let width = (width & 127).min(32 - offset);
+        if width == 0 { return None; }
+        self.right(offset)?.and(Self::low_mask(width as u8))
+    }
+
+    pub fn evaluate(self, value: u32) -> u32 {
+        ((value >> self.start) & Self::low_mask(self.width)) << self.shift
+    }
+}

+/// Independent private user-data encoding; no changes to L5's existing
+/// prefix or cache representation. 6 slot, 5 word, 5 start, 5 width-1,
+/// 5 shift, 2 flags = 28 bits. Remaining top nibble identifies the kind.
+#[derive(Clone, Copy, Debug, PartialEq, Eq)]
+pub struct L7SrdIndex {
+    pub heap: bool,
+    pub low30: bool,
+    pub bits: L7RootBits,
+}
+
+impl L7SrdIndex {
+    const PREFIX: u32 = 0x6000_0000;
+
+    pub fn encode(self) -> u32 {
+        let b = self.bits;
+        Self::PREFIX | u32::from(b.slot) | (u32::from(b.word) << 6)
+            | (u32::from(b.start) << 11) | (u32::from(b.width - 1) << 16)
+            | (u32::from(b.shift) << 21) | (u32::from(self.heap) << 26)
+            | (u32::from(self.low30) << 27)
+    }
+
+    pub fn decode(kind: u32) -> Option<Self> {
+        if kind & 0xf000_0000 != Self::PREFIX { return None; }
+        let b = L7RootBits {
+            slot: (kind & 63) as u8,
+            word: ((kind >> 6) & 31) as u8,
+            start: ((kind >> 11) & 31) as u8,
+            width: ((kind >> 16) & 31) as u8 + 1,
+            shift: ((kind >> 21) & 31) as u8,
+        };
+        if b.start + b.width > 32 || b.shift + b.width > 32 { return None; }
+        Some(Self { heap: kind & (1 << 26) != 0, low30: kind & (1 << 27) != 0, bits: b })
+    }
+
+    pub fn byte_offset(self, base: u32, offset: u32, value: u32) -> Option<u32> {
+        // Same fail-closed heap bounds policy as L5. Do not wrap a table
+        // base plus an index into a different heap slot.
+        base.checked_add(offset)?.checked_add(self.bits.evaluate(value))
+    }
+}
