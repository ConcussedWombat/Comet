+fn hair_uav_barrier(list: &HostList) {
+    // SAFETY: an open command list; a global UAV barrier has no resource
+    // reference to retain or release. No resource state is changed.
+    unsafe { list.list.ResourceBarrier(&[hair_uav_barrier_desc()]) };
+    static BARRIERS: AtomicU64 = AtomicU64::new(0);
+    let n = BARRIERS.fetch_add(1, Ordering::Relaxed) + 1;
+    if n.is_power_of_two() {
+        tracing::info!(barriers = n, "R1 hair UAV barriers recorded");
+    }

+/// G4: an independent census of actual failure sites, not the legacy
+/// classification inferred from the current pipeline state.
+#[cfg(all(windows, feature = "diagnostics"))]
+pub(crate) fn g4_enabled() -> bool {
+    static ON: OnceLock<bool> = OnceLock::new();
+    *ON.get_or_init(|| std::env::var_os("COMET_G4_DROP_CENSUS").is_some_and(|v| v == "1"))
+}
+
+#[cfg(all(windows, feature = "diagnostics"))]
+static G4_DROPS: Mutex<Option<HashMap<String, u64>>> = Mutex::new(None);
+
+#[cfg(all(windows, feature = "diagnostics"))]
+pub(crate) fn g4_note(this: This, reason: &str, missing: bool) {
+    if !g4_enabled() {
+        return;
+    }
+    let (stages, detail) = if this.is_null() {
+        ("no pipeline identity".to_owned(), String::new())
+    } else {
+        let p = pipeline(this);
+        let stages = p.shaders.iter().map(|s| format!("{} {}", s.stage, s.hash))
+            .collect::<Vec<_>>().join(" + ");
+        let detail = if missing {
+            skipped_shader(p).map(|h| format!("skipped shader {h}"))
+                .or_else(|| make::missing(&p.host))
+                .unwrap_or_else(|| "pipeline became ready after refusal".into())
+        } else {
+            String::new()
+        };
+        (stages, detail)
+    };
+    let key = format!("{reason} | {stages} | {detail}");
+    let mut counts = G4_DROPS.lock().unwrap_or_else(|e| e.into_inner());
+    *counts.get_or_insert_with(HashMap::new).entry(key).or_default() += 1;
+}
+
+#[cfg(all(windows, feature = "diagnostics"))]
+pub(crate) fn g4_take() -> Vec<(String, u64)> {
+    let mut counts: Vec<_> = G4_DROPS.lock().unwrap_or_else(|e| e.into_inner())
+        .take().into_iter().flatten().collect();
+    counts.sort_unstable_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
+    counts
+}
