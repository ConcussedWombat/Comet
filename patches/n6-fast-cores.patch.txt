+//! Optional preferred CPU sets for the layer's own wait/copy-back threads.
+use windows_sys::Win32::System::SystemInformation::{GetSystemCpuSetInformation, SYSTEM_CPU_SET_INFORMATION};
+use windows_sys::Win32::System::Threading::{GetCurrentProcess, GetCurrentThread, SetThreadSelectedCpuSets};
+
+fn fastest(sets: &[(u32, u8)]) -> Vec<u32> {
+    let Some(best) = sets.iter().map(|s| s.1).max() else { return Vec::new(); };
+    let chosen: Vec<_> = sets.iter().filter(|s| s.1 == best).map(|s| s.0).collect();
+    if chosen.len() == sets.len() { Vec::new() } else { chosen }
+}
+
+fn query() -> Vec<(u32, u8)> {
+    let mut size = 0u32;
+    // SAFETY: query the required byte count for the current process.
+    unsafe { GetSystemCpuSetInformation(std::ptr::null_mut(), 0, &mut size, GetCurrentProcess(), 0); }
+    if size == 0 { return Vec::new(); }
+    let mut storage = vec![0u64; (size as usize).div_ceil(8)];
+    let capacity = size;
+    // SAFETY: the aligned buffer holds at least capacity bytes.
+    if unsafe { GetSystemCpuSetInformation(storage.as_mut_ptr().cast(), capacity, &mut size, GetCurrentProcess(), 0) } == 0 || size > capacity {
+        return Vec::new();
+    }
+    let mut at = 0usize;
+    let mut out = Vec::new();
+    while at + size_of::<SYSTEM_CPU_SET_INFORMATION>() <= size as usize {
+        // SAFETY: records are aligned and the whole record header is in the buffer.
+        let info = unsafe { &*storage.as_ptr().cast::<u8>().add(at).cast::<SYSTEM_CPU_SET_INFORMATION>() };
+        let n = info.Size as usize;
+        if n < size_of::<SYSTEM_CPU_SET_INFORMATION>() || at + n > size as usize || !n.is_multiple_of(8) { return Vec::new(); }
+        // CPU-set records are the only type defined by this API currently.
+        let set = unsafe { &info.Anonymous.CpuSet };
+        // Do not assume a processor number is globally unique. This first
+        // port only targets single-group machines, e.g. the reference laptop.
+        if set.Group != 0 { return Vec::new(); }
+        out.push((set.Id, set.EfficiencyClass));
+        at += n;
+    }
+    out
+}
+
+pub(crate) fn prefer(what: &'static str) {
+    static ON: std::sync::OnceLock<bool> = std::sync::OnceLock::new();
+    if !*ON.get_or_init(|| std::env::var_os("COMET_FAST_CORES").is_some_and(|v| v == "1")) { return; }
+    let fast = fastest(&query());
+    if fast.is_empty() { return; }
+    // SAFETY: current thread and CPU set IDs returned by the OS.
+    let ok = unsafe { SetThreadSelectedCpuSets(GetCurrentThread(), fast.as_ptr(), fast.len() as u32) } != 0;
+    tracing::info!(what, selected = fast.len(), ok, "N6 fast-core preference (COMET_FAST_CORES=1)");
+}
+
+#[cfg(test)]
+#[test]
+fn selects_only_highest_efficiency_class() {
+    assert!(fastest(&[(1, 0), (2, 0)]).is_empty());
+    assert!(fastest(&[]).is_empty());
+    assert_eq!(fastest(&[(1, 0), (2, 8), (3, 8), (4, 4)]), vec![2, 3]);
+}
