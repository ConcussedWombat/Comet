+#[cfg(windows)]
+fn z3_join_ready(
+    held: &mut Vec<Held>,
+    lists: &mut Vec<crate::list::Detached>,
+    order: &mut Vec<usize>,
+) -> Vec<usize> {
+    let mut parts = Vec::new();
+    while !order.is_empty() {
+        let waits = held.iter().take_while(|h| matches!(h, Held::Value(_))).count();
+        let Some(Held::Run(next, next_order)) = held.get(waits) else { break };
+        if !z3_fits(parts.len() + 1, order.len(), next_order.len()) {
+            break;
+        }
+        Z3_CANDIDATES.fetch_add(1, Ordering::Relaxed);
+        // Do not predict values from earlier GPU work. Every intervening
+        // wait must already be met, and the joined work must not publish.
+        if !held[..waits].iter().all(|h| matches!(h, Held::Value(w) if w.met()))
+            || !order.iter().all(|&i| crate::host::z3_read_only(lists[i].host()))
+            || !next_order.iter().all(|&i| crate::host::z3_read_only(next[i].host()))
+        {
+            break;
+        }
+        // The inspected entries remain locked; drain only after all checks.
+        let Some(Held::Run(more, then)) = held.drain(..=waits).last() else { unreachable!() };
+        z3_append_order(order, &mut parts, lists.len(), then);
+        lists.extend(more);
+        Z3_JOINED.fetch_add(1, Ordering::Relaxed);
+    }
