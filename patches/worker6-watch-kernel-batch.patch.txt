+    #[cfg(feature = "diagnostics")]

+        use std::sync::atomic::Ordering::Relaxed;
+        let started = std::time::Instant::now();
+        let result = f();
+        WATCH_CALLS[kind].0.fetch_add(1, Relaxed);
+        WATCH_CALLS[kind].1.fetch_add(started.elapsed().as_micros() as u64, Relaxed);
+        return result;
+    }
+    let _ = kind;
+    f()
