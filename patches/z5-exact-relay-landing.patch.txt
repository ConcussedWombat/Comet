+/// Adapt VIctory's changed-dword put: adjacent changed bytes/dwords
+/// share a call, but never widen partial words or join unchanged gaps.
+/// In particular, never restore newer CPU bytes excluded by the merge.
+fn relay_landing_runs(was: &[u8], merged: &[u8], mut put: impl FnMut(usize, &[u8])) -> (usize, usize) {
+    let runs = changed_runs(merged, was);
+    let mut bytes = 0;
+    for &(at, n) in &runs {
+        put(at, &merged[at..at + n]);
+        bytes += n;
+    }
+    (runs.len(), bytes)
+}

+/// Retain existing guard notification and kernel-checked write path.
+fn relay_put(at: usize, merged: &[u8], was: Option<&[u8]>) -> usize {
+    crate::watch::writing(at, merged.len());
+    let Some(was) = was else {
+        crate::host::write_mapped(at as *mut u8, merged.as_ptr(), merged.len());
+        return merged.len();
+    };
+    let (runs, bytes) = relay_landing_runs(was, merged, |off, data| {
+        crate::host::write_mapped((at + off) as *mut u8, data.as_ptr(), data.len());
+    });
+    for (counter, n) in Z5_LANDING.iter().zip([1, runs, merged.len(), bytes]) {
+        counter.fetch_add(n as u64, Ordering::Relaxed);
+    }
+    bytes
+}
