+// The emitter writes the same fixed wave size into DXIL entry-point
+// property 11 and PSV0's MinWaveLaneCount/MaxWaveLaneCount. Read the
+// actual submitted container (including cache hits and R4 signed copies).
+// PSV0 starts with RuntimeInfoSize, followed by runtime info; the two
+// u32s are at offsets 16 and 20 within runtime info (all PSV versions).
+fn n2_declared_wave_size(psv: &[u8]) -> Result<Option<u32>, &'static str> {
+    fn word(bytes: &[u8], at: usize) -> Option<u32> {
+        Some(u32::from_le_bytes(bytes.get(at..at + 4)?.try_into().ok()?))
+    }
+    let size = word(psv, 0).ok_or("missing PSV0 runtime size")? as usize;
+    let info = psv.get(4..).and_then(|p| p.get(..size))
+        .ok_or("truncated PSV0 runtime info")?;
+    let min = word(info, 16).ok_or("missing PSV0 wave minimum")?;
+    let max = word(info, 20).ok_or("missing PSV0 wave maximum")?;
+    if min > max {
+        return Err("invalid PSV0 wave range");
+    }
+    // An unconstrained or ranged shader does not declare a single size.
+    Ok((min != 0 && min == max && max != u32::MAX).then_some(min))
+}

+fn n2_wave_census(shader: &Shader, bytes: &[u8], (min, max): (u32, u32)) {
+    let declared = comet_shader::Container::parse(bytes)
+        .map_err(|_| "invalid DXIL container")
+        .and_then(|c| {
+            let psv = c.part(comet_shader::FourCC::PSV0).ok_or("missing PSV0")?;
+            n2_declared_wave_size(psv.data)
+        });
+    let size = match declared {
+        Ok(Some(size)) if size < min || size > max => Some(size),
+        Ok(_) => return,
+        Err(_) => None,
+    };
+    // One line per unique stage/hash/declared mismatch, not per PSO.
+    type Key = (&'static str, String, Option<u32>);
+    static SEEN: std::sync::OnceLock<std::sync::Mutex<std::collections::HashSet<Key>>> =
+        std::sync::OnceLock::new();
+    let first = SEEN.get_or_init(Default::default)
+        .lock().unwrap_or_else(|e| e.into_inner())
+        .insert((shader.stage, shader.hash.clone(), size));
+    if !first {
+        return;
+    }
+    match declared {
+        Ok(Some(declared)) => tracing::warn!(
+            hash = %shader.hash, stage = shader.stage, declared, min, max,
+            "N2 unsupported shader wave size"
+        ),
+        Err(reason) => tracing::warn!(
+            hash = %shader.hash, stage = shader.stage, reason,
+            "N2 shader wave size unreadable"
+        ),
+        Ok(None) => {}
+    }
+}
