+/// A mark's title-page intersection with the existing declared ranges.
+/// Missing certification, malformed ranges or an out-of-range bit must
+/// not turn a dirty copy into a clean one. The caller falls back to full
+/// comparison when this returns None.
+fn marked_spans(run: MarkRun, words: &[u32], start: usize, len: usize,
+                declared: &[(usize, usize)]) -> Option<Vec<(usize, usize)>> {
+    let end = start.checked_add(len)?;
+    if len == 0 || declared.iter().any(|&(a, e)| a < start || a >= e || e > end) {
+        return None;
+    }
+    let first_page = start / PAGE;
+    if end.div_ceil(PAGE).checked_sub(first_page)? != run.pages as usize { return None; }
+    let mut spans: Vec<(usize, usize)> = Vec::new();
+    for i in 0..run.pages {
+        let bit = run.base.checked_add(i)?;
+        if words.get((bit / 32) as usize)? & (1u32 << (bit % 32)) == 0 { continue; }
+        let p = first_page.checked_add(i as usize)?.checked_mul(PAGE)?;
+        let a = p.max(start);
+        let e = p.checked_add(PAGE)?.min(end);
+        let mut covered = false;
+        for &(lo, hi) in declared {
+            let (lo, hi) = (lo.max(a), hi.min(e));
+            if lo < hi { spans.push((lo, hi)); covered = true; }
+        }
+        if !covered { return None; }
+    }
+    spans.sort_unstable();
+    let mut union: Vec<(usize, usize)> = Vec::new();
+    for (a, e) in spans {
+        if let Some(last) = union.last_mut()
+            && a <= last.1
+        {
+            last.1 = last.1.max(e);
+        } else { union.push((a, e)); }
+    }
+    Some(union)
+}

+/// Read only after the fence for the whole submission has completed.
+/// An empty selected span list is meaningful only with a certificate.

+pub(crate) fn finish_marked_pages(

+    certified: &BTreeMap<u64, bool>,
+) -> MarkSelection {
+    let mut selected = BTreeMap::new();
+    let Some(b) = mark_bitmap().get().and_then(Option::as_ref) else { return selected; };
+    if !marked_sync() { return selected; }
+    // SAFETY: the entire submission fence completed, no other submission
+    // can run under the caller's lock, and the bitmap lives for good.
+    let words = unsafe { std::slice::from_raw_parts(b.data as *const u32, MARK_WORDS) };
+    if words[(MARK_OVERFLOW / 32) as usize] & (1 << (MARK_OVERFLOW % 32)) != 0 {
+        MARK_STATS[3].fetch_add(1, Ordering::Relaxed);
+        return selected;
+    }
+    let runs = MARK_RUNS.lock().unwrap_or_else(|e| e.into_inner());
+    for (shadow, declared) in back {
+        if certified.get(&shadow.id()) != Some(&true) || shadow.carded() || shadow.place.is_some() {
+            continue;
+        }
+        let Some(&run) = runs.owners.get(&shadow.id()) else { continue; };
+        let Some(spans) = marked_spans(run, words, shadow.start as usize, shadow.len as usize, declared) else {
+            MARK_STATS[3].fetch_add(1, Ordering::Relaxed);
+            continue;
+        };
+        // The existing snapshots are owner-local, and no guard saves are
+        // armed by this stage. Consume only this owner's selection before
+        // any title writes (940d32f ordering); other copies keep theirs.
+        let mut publication = crate::watch::MarkedPublication::new(shadow.id(), spans);
+        let Some(spans) = publication.unarm(shadow.id()) else { continue; };

+    }
