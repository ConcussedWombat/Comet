+        crate::capture::op(list_id, || format!("G4-WIND stage={} hash={hash} stores={} name_match={} indexed={:?} root_srvs={:?} root_uavs={:?}",

+        let entry = |slot: u8, bytes: u64, kind: &str| {
+            crate::capture::op(list_id, || {
+                let Some(Root::Table(table)) = self.roots_ref(compute).get(usize::from(slot)) else {
+                    return format!("G4-WIND-VIEW hash={hash} slot={slot} bytes={bytes} kind={kind} no_title_table");
+                };
+                let found = (|| {
+                    let heap = self.heap_of(*table)?;
+                    if heap.samplers() || (*table - heap.start) % 32 != 0 {
+                        return None;
+                    }
+                    let handle = table.checked_add(bytes)?;
+                    let title_index = heap.index(handle)?;
+                    let host = heap.host()?;
+                    let index = if let Some(&(_, base)) = made.iter().find(|m| m.0 == slot) {
+                        if base == u32::MAX { return None; }
+                        (base / 32).checked_add(u32::try_from(bytes / 32).ok()?)?
+                    } else {
+                        title_index
+                    };
+                    let view = crate::descriptor::view(handle);
+                    Some(format!("title_heap={:#x} title_index={title_index} host_index={index} title_view={view:x?} backing={}",
+                        heap.start, host.g4_backing(index)))
+                })();
+                format!("G4-WIND-VIEW hash={hash} slot={slot} bytes={bytes} kind={kind} {}", found.unwrap_or_else(|| "unresolved".into()))
+            });
+        };
+        for u in &t.used {
+            if u.root == RootType::Table && matches!(u.kind, ResourceType::Srv | ResourceType::Uav) {
+                entry(u.slot, u64::from(u.table_offset_dwords) * 4, if u.kind == ResourceType::Srv { "srv" } else { "uav" });
+            }
+        }

+            if index.shift >= 32 {

+            }

+            };

+            };

+        }
