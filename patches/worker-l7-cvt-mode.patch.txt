+    /// Reads an IEEE-754 binary64 operand as `(lo, hi)` 32-bit words, applying
+    /// `abs` and `neg` input modifiers on operand `k`.
+    fn l7_f64_pair(&mut self, inst: &Inst, k: usize) -> Result<(V, V), Why> {
+        let Fields::Valu(v) = &inst.fields else {
+            unreachable!()
+        };
+        if v.sdwa.is_some() || v.dpp.is_some() {
+            return Err("64-bit operand modifiers");
+        }
+        let op = *v.src.get(k).ok_or("a missing operand")?;
+        let pair = |l: &mut Self, r: Reg, r1: Reg| -> Result<(V, V), Why> {
+            let lo = l.i32_of(inst.read(r).ok_or("an unread operand")?)?;
+            let hi = l.i32_of(inst.read(r1).ok_or("an unread operand")?)?;
+            Ok((lo, hi))
+        };
+        let (lo, mut hi) = match op {
+            Operand::Vector { reg, .. } => pair(self, Reg::V(reg), Reg::V(reg + 1))?,
+            Operand::Scalar { code, .. } if code != NULL => {
+                pair(self, Reg::S(code), Reg::S(code + 1))?
+            }
+            Operand::Const(k @ 128..=208) => {
+                let bits = crate::resource::inline(k).ok_or("a 64-bit inline operand")?;
+                (self.s.m.i32(bits), self.s.m.i32(if (bits as i32) < 0 { u32::MAX } else { 0 }))
+            }
+            Operand::Scalar { code: NULL, .. } => (self.s.m.i32(0), self.s.m.i32(0)),
+            Operand::Const(c @ 240..=248) => {
+                const INLINE_F64: [f64; 9] = [
+                    0.5,
+                    -0.5,
+                    1.0,
+                    -1.0,
+                    2.0,
+                    -2.0,
+                    4.0,
+                    -4.0,
+                    1.0 / std::f64::consts::TAU,
+                ];
+                let bits = INLINE_F64[(c - 240) as usize].to_bits();
+                (self.s.m.i32(bits as u32), self.s.m.i32((bits >> 32) as u32))
+            }
+            // GFX10 places a 32-bit literal in the high 32 bits of a 64-bit float operand.
+            Operand::Literal(x) => (self.s.m.i32(0), self.s.m.i32(x)),
+            _ => return Err("a 64-bit float operand"),
+        };
+        let c = self.cur;
+        if v.abs >> k & 1 != 0 {
+            let mask = self.s.m.i32(0x7FFF_FFFF);
+            hi = self.s.m.bin(c, Bin::And, hi, mask);
+        }
+        if v.neg >> k & 1 != 0 {
+            let sign = self.s.m.i32(0x8000_0000);
+            hi = self.s.m.bin(c, Bin::Xor, hi, sign);
+        }
+        Ok((lo, hi))
+    }

+    /// Rounded integer significand, with all discarded bits included in the
+    /// rounding decision. Shift is in [29,63]; no out-of-range DXIL shifts.
+    fn l7_round_sig(&mut self, sig: V, shift: V, away: V, nearest: bool) -> V {
+        let c = self.cur;
+        let i64t = self.s.m.int(64);
+        let zero = self.s.m.int_const(i64t, 0);
+        let one = self.s.m.int_const(i64t, 1);
+        let shift = self.s.m.cast(c, Cast::ZExt, shift, i64t);
+        let q = self.s.m.bin(c, Bin::LShr, sig, shift);
+        let unit = self.s.m.bin(c, Bin::Shl, one, shift);
+        let mask = self.s.m.bin(c, Bin::Sub, unit, one);
+        let rem = self.s.m.bin(c, Bin::And, sig, mask);
+        let inc = if nearest {
+            let half = self.s.m.bin(c, Bin::LShr, unit, one);
+            let above = self.s.m.cmp(c, pred::IUGT, rem, half);
+            let tie = self.s.m.cmp(c, pred::EQ, rem, half);
+            let odd = self.s.m.bin(c, Bin::And, q, one);
+            let odd = self.s.m.cmp(c, pred::NE, odd, zero);
+            let tie_odd = self.s.m.bin(c, Bin::And, tie, odd);
+            self.s.m.bin(c, Bin::Or, above, tie_odd)
+        } else {
+            let inexact = self.s.m.cmp(c, pred::NE, rem, zero);
+            self.s.m.bin(c, Bin::And, away, inexact)
+        };
+        let inc = self.s.m.select(c, inc, one, zero);
+        let rounded = self.s.m.bin(c, Bin::Add, q, inc);
+        self.s.m.cast(c, Cast::Trunc, rounded, self.s.t.i32)
+    }

+    fn l7_narrow_f64(&mut self, lo: V, hi: V, mode: u32, omod: u8, clamp: bool) -> V {
+        let c = self.cur;
+        let i64t = self.s.m.int(64);
+        self.s.flags |= flags::INT64;
+        let [zero, sign_mask, k20, k7ff, frac_mask, k896, k897, k926, k29, k63,
+             k1150, k23, inf, max, k254, k255, k24, mant_mask] =
+            [0, 0x8000_0000, 20, 0x7ff, 0x000f_ffff, 896, 897, 926, 29, 63,
+             1150, 23, 0x7f80_0000, 0x7f7f_ffff, 254, 255, 24, 0x007f_ffff]
+                .map(|n| self.s.m.i32(n));
+        let sign = self.s.m.bin(c, Bin::And, hi, sign_mask);
+        let negative = self.s.m.cmp(c, pred::NE, sign, zero);
+        let nearest = mode & 3 == 0;
+        let away = match mode & 3 {
+            1 => self.s.m.cmp(c, pred::EQ, sign, zero),
+            2 => negative,
+            _ => self.s.m.bool(false),
+        };
+        let exp = self.s.m.bin(c, Bin::LShr, hi, k20);
+        let exp = self.s.m.bin(c, Bin::And, exp, k7ff);
+        let fraction_hi = self.s.m.bin(c, Bin::And, hi, frac_mask);
+        let fraction = self.join64(lo, fraction_hi);
+        let z64 = self.s.m.int_const(i64t, 0);
+        let frac_nonzero = self.s.m.cmp(c, pred::NE, fraction, z64);
+        let exp_zero = self.s.m.cmp(c, pred::EQ, exp, zero);
+        let hidden = self.s.m.int_const(i64t, 1i64 << 52);
+        let hidden = self.s.m.select(c, exp_zero, z64, hidden);
+        let sig = self.s.m.bin(c, Bin::Or, fraction, hidden);
+        let sig = if mode & 0x40 == 0 {
+            self.s.m.select(c, exp_zero, z64, sig)
+        } else { sig };
+        let overflow_mag = if nearest { inf } else { self.s.m.select(c, away, inf, max) };
+        let active_omod = omod != 0 && mode & 0x220 == 0;
+
+        let mag = if mode & 0x20 != 0 {
+            // Output denormals preserved: one rounding to binary32, with
+            // the subnormal quantum 2^-149. OMOD is ignored in this mode.
+            let normal = self.s.m.cmp(c, pred::IUGE, exp, k897);
+            let shift = self.s.m.bin(c, Bin::Sub, k926, exp);
+            let shift = self.s.m.select(c, normal, k29, shift);
+            let too_far = self.s.m.cmp(c, pred::IUGT, shift, k63);
+            let shift = self.s.m.select(c, too_far, k63, shift);
+            let q = self.l7_round_sig(sig, shift, away, nearest);
+            let biased = self.s.m.bin(c, Bin::Sub, exp, k897);
+            let biased = self.s.m.bin(c, Bin::Shl, biased, k23);
+            let norm = self.s.m.bin(c, Bin::Add, biased, q);
+            let mag = self.s.m.select(c, normal, norm, q);
+            let overflow = self.s.m.cmp(c, pred::IUGT, exp, k1150);
+            self.s.m.select(c, overflow, overflow_mag, mag)
+        } else {

+//! Static MODE facts at *machine* instruction offsets. Analyze the original
+//! CFG, not a host rewrite which can remove EXEC skips. A fact is valid on
+//! every wave-level path to that offset; unknown bits never become zero.
+//!
+//! RDNA2 ISA tables 4, 16, 24: HW_REG_MODE=1; hwreg immediates carry
+//! id[5:0], offset[10:6], (width-1)[15:11]. MODE[3:0]=round,
+//! [7:4]=denorm, [8]=DX10_CLAMP, [9]=IEEE. Other MODE bits are not
+//! modeled (e.g. sticky LOD_CLAMPED can change through vector instructions).
+
+use std::collections::{HashMap, VecDeque};
+
+use crate::gfx10::{Fields, NULL, Operand};
+use crate::ir::{self, Inst, Program, Reg};
+use crate::native::NativeProgram;

+const FLOAT_BITS: u32 = 0x3ff;
+
+#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
+struct Bits {
+    known: u32,
+    value: u32,
+}
+
+impl Bits {
+    fn new(known: u32, value: u32) -> Self {
+        Self { known, value: value & known }
+    }
+
+    fn constant(value: u32) -> Self {
+        Self::new(u32::MAX, value)
+    }
+
+    fn exact(self) -> Option<u32> {
+        (self.known == u32::MAX).then_some(self.value)
+    }
+
+    fn meet(self, other: Self) -> Self {
+        Self::new(self.known & other.known & !(self.value ^ other.value), self.value)
+    }
+
+    fn and(self, other: Self) -> Self {
+        let known = (self.known & other.known)
+            | (self.known & !self.value) | (other.known & !other.value);
+        Self::new(known, self.value & other.value)
+    }
+
+    fn or(self, other: Self) -> Self {
+        let known = (self.known & other.known) | self.value | other.value;
+        Self::new(known, self.value | other.value)
+    }
+
+    fn xor(self, other: Self) -> Self {
+        Self::new(self.known & other.known, self.value ^ other.value)
+    }
+
+    fn not(self) -> Self {
+        Self::new(self.known, !self.value)
+    }
+
+    fn shl(self, n: u32) -> Self {
+        let n = n & 31;
+        Self::new((self.known << n) | !(u32::MAX << n), self.value << n)
+    }
+
+    fn shr(self, n: u32) -> Self {
+        let n = n & 31;
+        Self::new((self.known >> n) | !(u32::MAX >> n), self.value >> n)
+    }
+
+    fn field(self, offset: u32, width: u32) -> Self {
+        let mask = u32::MAX >> (32 - width);
+        Self::new((self.known >> offset) | !mask, (self.value >> offset) & mask)
+    }
+
+    fn write(self, offset: u32, width: u32, data: Self) -> Self {
+        let mask = (u32::MAX >> (32 - width)) << offset;
+        Self::new((self.known & !mask) | ((data.known << offset) & mask),
+                  (self.value & !mask) | ((data.value << offset) & mask))
+    }
+}

+/// GFX10 PGM_RSRC1 fields are not MODE's bit positions.
+pub(super) fn from_rsrc1(rsrc1: u32) -> u32 {
+    ((rsrc1 >> 12) & 0xff) | ((rsrc1 >> 13) & 0x100) | ((rsrc1 >> 14) & 0x200)

+fn hwreg(imm: u16) -> Option<(u32, u32)> {
+    if imm & 63 != 1 {
+        return None;
+    }
+    let offset = u32::from((imm >> 6) & 31);
+    let width = u32::from(imm >> 11) + 1;
+    (offset + width <= 32).then_some((offset, width))
+}

+pub(super) fn mode_register(imm: u16) -> bool {
+    imm & 63 == 1
+}

+#[derive(Clone, PartialEq, Eq)]
+struct State {
+    mode: Bits,
+    regs: [Bits; 256],
+}
+
+impl State {
+    fn initial(rsrc1: Option<u32>) -> Self {
+        Self {
+            mode: rsrc1.map_or_else(Bits::default, |r| Bits::new(FLOAT_BITS, from_rsrc1(r))),
+            regs: [Bits::default(); 256],
+        }
+    }
+
+    fn meet(&mut self, other: &Self) -> bool {
+        let mut changed = false;
+        let mode = self.mode.meet(other.mode);
+        changed |= mode != self.mode;
+        self.mode = mode;
+        for (a, b) in self.regs.iter_mut().zip(other.regs) {
+            let joined = a.meet(b);
+            changed |= joined != *a;
+            *a = joined;
+        }
+        changed
+    }
+
+    fn operand(&self, op: Option<&Operand>) -> Bits {
+        match op {
+            Some(Operand::Literal(x)) => Bits::constant(*x),
+            Some(Operand::Const(code)) => crate::resource::inline(*code)
+                .map_or_else(Bits::default, Bits::constant),
+            Some(Operand::Scalar { code: NULL, .. }) => Bits::constant(0),
+            Some(Operand::Scalar { code, .. }) => self.regs[usize::from(*code)],
+            _ => Bits::default(),
+        }
+    }
+
+    fn getreg(&self, inst: &Inst) -> Option<Bits> {
+        if inst.name() != "s_getreg_b32" {
+            return None;
+        }
+        let Fields::Sop { imm: Some(imm), .. } = inst.fields else { return None };
+        let (offset, width) = hwreg(imm)?;
+        Some(self.mode.field(offset, width))
+    }
+
+    fn step(&mut self, inst: &Inst) {
+        if let Fields::Sopp { imm } = inst.fields {
+            match inst.name() {
+                "s_round_mode" => {
+                    self.mode = self.mode.write(0, 4, Bits::constant(u32::from(imm & 15)));
+                }
+                "s_denorm_mode" => {
+                    self.mode = self.mode.write(4, 4, Bits::constant(u32::from(imm & 15)));
+                }
+                _ => {}
+            }
+        }
+        let mut dest = None;
+        if let Fields::Sop { dst, src, imm } = &inst.fields {
+            let a = self.operand(src.first());
+            let b = self.operand(src.get(1));
+            match inst.name() {
+                "s_setreg_b32" | "s_setreg_imm32_b32" => {
+                    if let Some(imm) = *imm && mode_register(imm) {
+                        self.mode = match hwreg(imm) {
+                            Some((offset, width)) => {
+                                let updated = self.mode.write(offset, width, a);
+                                Bits::new(updated.known & FLOAT_BITS, updated.value)
+                            }
+                            None => Bits::default(),
+                        };
+                    }
+                }
+                _ => {}
+            }
+            let value = match inst.name() {
+                "s_getreg_b32" => self.getreg(inst).unwrap_or_default(),
+                "s_mov_b32" => a,
+                "s_movk_i32" => Bits::constant(i32::from(imm.unwrap_or(0) as i16) as u32),
+                "s_not_b32" => a.not(),
+                "s_and_b32" => a.and(b),
+                "s_andn2_b32" => a.and(b.not()),
+                "s_or_b32" => a.or(b),
+                "s_orn2_b32" => a.or(b.not()),
+                "s_xor_b32" => a.xor(b),
+                "s_lshl_b32" => b.exact().map_or_else(Bits::default, |n| a.shl(n)),
+                "s_lshr_b32" => b.exact().map_or_else(Bits::default, |n| a.shr(n)),
+                "s_add_u32" | "s_add_i32" => a.exact().zip(b.exact())
+                    .map_or_else(Bits::default, |(x, y)| Bits::constant(x.wrapping_add(y))),
+                "s_sub_u32" | "s_sub_i32" => a.exact().zip(b.exact())
+                    .map_or_else(Bits::default, |(x, y)| Bits::constant(x.wrapping_sub(y))),
+                _ => Bits::default(),
+            };
+            if let Some(Operand::Scalar { code, .. }) = dst {
+                dest = Some((*code, value));
+            }
+        }
+        // Unsupported producers lose their facts. Do not assume a user SGPR,
+        // memory load, lane read, SCC-dependent move, or arithmetic is constant.
+        for &(reg, _) in &inst.writes {
+            if let Reg::S(code) = reg {
+                self.regs[usize::from(code)] = Bits::default();
+            }
+        }
+        if let Some((code, value)) = dest
+            && inst.write(Reg::S(code)).is_some()
+            && code != NULL
+        {
+            self.regs[usize::from(code)] = value;
+        }
+    }
+}

+#[derive(Default)]
+pub(super) struct Analysis {
+    before: HashMap<usize, Bits>,
+    reads: HashMap<usize, Bits>,
+}
+
+impl Analysis {
+    pub(super) fn new(native: &NativeProgram) -> Self {
+        let Ok(p) = ir::build(&native.patched_code(), native.wave64, native.vgprs) else {
+            return Self::default();
+        };
+        Self::analyze(&p, native.rsrc1)
+    }
+
+    fn analyze(p: &Program, rsrc1: Option<u32>) -> Self {
+        let mut incoming: Vec<Option<State>> = vec![None; p.blocks.len()];
+        if incoming.is_empty() { return Self::default(); }
+        incoming[0] = Some(State::initial(rsrc1));
+        let mut pending = VecDeque::from([0]);
+        while let Some(b) = pending.pop_front() {
+            let mut state = incoming[b].as_ref().expect("reachable").clone();
+            for inst in &p.blocks[b].insts {
+                state.step(inst);
+            }
+            for to in p.blocks[b].exit.successors() {
+                let changed = match &mut incoming[to] {
+                    Some(old) => old.meet(&state),
+                    slot @ None => {
+                        *slot = Some(state.clone());
+                        true
+                    }
+                };
+                if changed {
+                    pending.push_back(to);
+                }
+            }
+        }
+        // Collect only after the fixed point. IR may contain duplicate blocks
+        // for irreducible loops; merge every copy at its original PC too.
+        let mut out = Self::default();
+        let note = |map: &mut HashMap<usize, Bits>, pc, fact| {
+            map.entry(pc).and_modify(|old| *old = old.meet(fact)).or_insert(fact);
+        };
+        for (b, block) in p.blocks.iter().enumerate() {
+            let Some(mut state) = incoming[b].take() else { continue };
+            for inst in &block.insts {
+                let pc = inst.machine.offset;
+                if inst.name() == "v_cvt_f32_f64" {
+                    note(&mut out.before, pc, state.mode);
+                }
+                if let Some(data) = state.getreg(inst) {
+                    note(&mut out.reads, pc, data);
+                }
+                state.step(inst);
+            }
+        }
+        out
+    }
+
+    /// Only the bits that can affect this instruction's result are needed.
+    /// Conversion always quiets NaN. F64 input denorms can produce a nonzero
+    /// binary32 only with directed rounding and preserved f32 output denorms.
+    /// IEEE matters for OMOD only, DX10_CLAMP for clamp only.
+    pub(super) fn conversion(&self, pc: usize, omod: u8, clamp: bool) -> Option<u32> {
+        let fact = self.before.get(&pc)?;
+        let mut required = 0x23; // round32 and f32 output denorm
+        if fact.known & required != required {
+            return None;
+        }
+        let mode = fact.value;
+        if mode & 0x20 != 0 && matches!(mode & 3, 1 | 2) {
+            required |= 0x40; // f64 input denorm
+        }
+        if omod != 0 && mode & 0x20 == 0 {
+            required |= 0x200;
+        }
+        if clamp {
+            required |= 0x100;
+        }
+        (fact.known & required == required).then_some(mode)
+    }
+
+    pub(super) fn getreg(&self, pc: usize) -> Option<u32> {
+        self.reads.get(&pc)?.exact()
+    }
+
+    pub(super) fn trace(&self, pc: usize) -> Option<(u32, u32)> {
+        self.before.get(&pc).map(|b| (b.known, b.value))
+    }
+}

+    const CVT: u32 = 0x7e00_1f01; // v_cvt_f32_f64 v0, v[1:2]
+    const END: u32 = 0xbf81_0000;
+    const fn hw(offset: u32, width: u32) -> u32 {
+        1 | (offset << 6) | ((width - 1) << 11)
+    }
+    const fn get(s: u32, offset: u32, width: u32) -> u32 {
+        0xb800_0000 | (0x12 << 23) | (s << 16) | hw(offset, width)
+    }
+    const fn set(s: u32, offset: u32, width: u32) -> u32 {
+        0xb800_0000 | (0x13 << 23) | (s << 16) | hw(offset, width)
+    }
+    const fn setimm(offset: u32, width: u32) -> u32 {
+        0xb800_0000 | (0x15 << 23) | hw(offset, width)
+    }
+
+    fn analyze(words: &[u32], rsrc1: Option<u32>) -> Analysis {
+        let code: Vec<_> = words.iter().flat_map(|w| w.to_le_bytes()).collect();
+        let p = ir::build(&code, false, 8).expect("IR");
+        Analysis::analyze(&p, rsrc1)
+    }

+    #[test]
+    fn l7_mode_rsrc1_fields_and_local_updates() {
+        let initial = 0x00ac_0000; // FLOAT_MODE c0, DX10_CLAMP=1, IEEE=1
+        assert_eq!(from_rsrc1(initial), 0x3c0);
+        let words = [CVT, 0xbf8c_0000, // s_waitcnt (dropped)
+                     0xbfa4_0002, CVT, // s_round_mode 2
+                     0xbfa5_000f, CVT, // s_denorm_mode 15
+                     END];
+        let a = analyze(&words, Some(initial));
+        assert_eq!(a.conversion(0, 0, false), Some(0x3c0));
+        assert_eq!(a.conversion(12, 0, false), Some(0x3c2));
+        assert_eq!(a.conversion(20, 0, false), Some(0x3f2));
+    }
+
+    #[test]
+    fn l7_mode_saved_field_restore_and_unrelated_register() {
+        let initial = 0x00ac_0000;
+        let words = [get(4, 0, 4), setimm(0, 2), 3, CVT,
+                     set(4, 0, 4), CVT,
+                     // TRAPSTS write (id 3): not MODE.
+                     (setimm(0, 4) & !63) | 3, 0, CVT, END];
+        let a = analyze(&words, Some(initial));
+        assert_eq!(a.getreg(0), Some(0));
+        assert_eq!(a.conversion(12, 0, false), Some(0x3c3));
+        assert_eq!(a.conversion(20, 0, false), Some(0x3c0));
+        assert_eq!(a.conversion(32, 0, false), Some(0x3c0));
+    }
+
+    #[test]
+    fn l7_mode_unknown_bits_can_be_overwritten() {
+        let words = [CVT, setimm(0, 8), 0xf1, CVT, set(2, 0, 2),
+                     CVT, setimm(0, 2), 3, CVT, END];
+        let a = analyze(&words, None);
+        assert_eq!(a.conversion(0, 0, false), None);
+        assert_eq!(a.conversion(12, 0, false), Some(0xf1));
+        assert_eq!(a.conversion(20, 0, false), None);
+        assert_eq!(a.conversion(32, 0, false), Some(0xf3));
+    }

+    fn l7_mode_control_flow_join_does_not_guess() {
+        // s_cbranch_scc0 +2 (skip a mode write and the jump).
+        let words = [0xbf84_0002, 0xbfa4_0001, 0xbf82_0001,
+                     0xbfa4_0002, CVT, setimm(0, 4), 0, CVT, END];
+        let a = analyze(&words, Some(0x000c_0000));
+        assert_eq!(a.conversion(16, 0, false), None);
+        assert_eq!(a.conversion(28, 0, false), Some(0xc0));
