+/// Returns the actual overlapping half-open title interval, not a shadow
+/// identity: distinct host copies of aliased title memory can conflict.
+fn g5_pending_overlap(pending: &[(usize, usize)], at: usize, len: usize) -> Option<(usize, usize)> {
+    let end = at.saturating_add(len);
+    if at >= end {
+        return None;
+    }
+    let i = pending.partition_point(|r| r.1 <= at);
+    let &(lo, hi) = pending.get(i)?;
+    (lo < end).then_some((lo.max(at), hi.min(end)))
+}
+
+fn g5_copyback_conflict(
+    pending: &[(usize, usize)],
+    runs: &[&HostList],
+) -> Option<(&'static str, usize, usize, usize)> {
+    if pending.is_empty() {
+        return None;
+    }
+    for list in runs {
+        for step in &list.deferred().copies {
+            let (kind, shadow, at, len) = match step {
+                Step::Refresh(shadow, at, len) => ("refresh", shadow, *at, *len),
+                Step::GpuWrite(shadow, at, len) => ("write", shadow, *at, *len),
+                _ => continue,
+            };
+            if let Some((lo, hi)) = g5_pending_overlap(pending, at, len) {
+                return Some((kind, std::sync::Arc::as_ptr(shadow) as usize, lo, hi));
+            }
+        }
+    }
+    None
+}
