+    /// Log exact root-derived indices for ordinary (non-relocated) tables.
+    /// This observes binding candidates, not execution or sampled pixels.
+    fn w3_trace_indexed_textures(
+        &self, p: &HostPipeline, hash: &str, indexed: &[RootIndexedView], made: &[(u8, u32)],
+    ) {
+        for index in indexed {
+            if made.iter().any(|&(slot, _)| slot == index.table_slot) {
+                continue;
+            }
+            let Some(Root::Constants(values)) = self.roots_ref(false).get(usize::from(index.constant_slot)) else {
+                continue;
+            };
+            let Some(&constant) = values.get(index.constant_dword as usize) else {
+                continue;
+            };
+            let Some(handle) = self.hair_selected_handle(index, false) else {
+                continue;
+            };
+            let Some(heap) = self.heap_of(handle) else { continue };
+            let (Some(host), Some(title_index)) = (heap.host(), heap.index(handle)) else { continue };
+            let Some(view) = crate::descriptor::view(handle) else { continue };
+            if !crate::descriptor::host::is_texture_view(&view) {
+                continue;
+            }
+            let in_host_heap = title_index < host.count();
+            let backed = in_host_heap && host.slot_has_resource(title_index);
+            let frame = crate::frame::current();
+            // Bound volume per frame, with an exact shader filter for
+            // repeated candidates or late streaming materials.
+            static LIMIT: std::sync::Mutex<(u64, u32)> = std::sync::Mutex::new((u64::MAX, 0));
+            let mut limit = LIMIT.lock().unwrap_or_else(|e| e.into_inner());
+            if limit.0 != frame {
+                *limit = (frame, 0);
+            }
+            if limit.1 >= 128 {
+                return;
+            }
+            limit.1 += 1;
+            drop(limit);
+            tracing::info!(frame, hash, pipeline = %p.name, table_slot = index.table_slot,
+                constant_slot = index.constant_slot, constant_dword = index.constant_dword,
+                constant, relative = selected_view_offset(index, constant), title_index,
+                in_host_heap, backed, ?view, "W3 exact indexed texture bind candidate");
+        }
+    }
