+    /// Adapted from fxceless/victory-fork c96b4f2 (see CREDITS.md).
+    fn m4_bake_preflight(&self, list: &HostList, p: &HostPipeline, compute: bool) {
+        if compute || p.compute {
+            return;
+        }
+        static MODES: std::sync::OnceLock<(bool, bool)> = std::sync::OnceLock::new();
+        let &(enabled, trace) = MODES.get_or_init(|| {
+            (
+                std::env::var_os("COMET_M4_BAKE_PREFLIGHT").is_some_and(|v| v == "1"),
+                std::env::var_os("COMET_M4_BAKE_TRACE").is_some_and(|v| v == "1"),
+            )
+        });
+        if !enabled && !trace {
+            return;
+        }
+        let Some(stage) = p.stages.iter().find(|s| {
+            matches!(s.shader.hash.as_str(),
+                "c0a7fe77b1862e10" | "0d1575a396dfe513" |
+                "b7cfd8dda91ebb6b" | "280967ad158cee0b")
+        }) else {
+            return;
+        };
+        let frame = crate::frame::current();
+        if trace {
+            tracing::info!(frame, hash = %stage.shader.hash, pipeline = %p.name,
+                "M4 decoration bake binding reached");
+        }
+        if !enabled {
+            return;
+        }
+        let Some(heap) = self.heaps.iter().filter(|&&h| h != 0)
+            .map(|&h| crate::descriptor::heap(h as This))
+            .find(|h| !h.samplers())
+        else {
+            return;
+        };
+        let Some(host) = heap.host() else {
+            return;
+        };
+        // No extra pass budget, and at most one extra queue walk per
+        // frame. The usual heap_buffers path shares the same byte cap.
+        static FRAME: AtomicU64 = AtomicU64::new(u64::MAX);
+        if FRAME.swap(frame, Ordering::Relaxed) == frame {
+            return;
+        }
+        let before = HEAP_TEXTURES.load(Ordering::Relaxed);
+        self.heap_textures(list, heap, host);
+        tracing::info!(frame, hash = %stage.shader.hash,
+            uploaded_before = before,
+            uploaded_after = HEAP_TEXTURES.load(Ordering::Relaxed),
+            waiting = HEAP_TEXTURES_WAITING.load(Ordering::Relaxed),
+            "M4 decoration bake texture preflight recorded");
+    }
