+//! Process isolation for tests that exercise the runtime's one-title globals
+//! or native D3D12. No production code or environment is changed.
+use std::process::{Command, Output};
+use std::sync::Mutex;
+
+// Keep one native test child at a time on the physical adapter. Pure unit
+// tests still run on libtest's ordinary parallel workers in the parent.
+static NATIVE: Mutex<()> = Mutex::new(());
+const CHILD: &str = "COMET_V5_ISOLATED_TEST";
+
+fn test_name(module: &str, name: &str) -> String {
+    let (_, module) = module.split_once("::").expect("nested test module");
+    format!("{module}::{name}")
+}
+
+pub(crate) fn in_child(module: &str, name: &str) -> bool {
+    let Some(child) = std::env::var_os(CHILD) else { return false };
+    assert_eq!(child, std::ffi::OsStr::new(&test_name(module, name)));
+    // SAFETY: this is a dedicated test child, before any native test work.
+    // Suppress Windows error-reporting UI even for a genuine access violation.
+    use windows_sys::Win32::System::Diagnostics::Debug::{
+        SEM_FAILCRITICALERRORS, SEM_NOGPFAULTERRORBOX, SEM_NOOPENFILEERRORBOX, SetErrorMode,
+    };
+    unsafe {
+        SetErrorMode(SEM_FAILCRITICALERRORS | SEM_NOGPFAULTERRORBOX | SEM_NOOPENFILEERRORBOX);
+    }
+    true
+}
+
+pub(crate) fn run_child(module: &str, name: &str) -> Output {
+    let name = test_name(module, name);
+    let _serial = NATIVE.lock().unwrap_or_else(|e| e.into_inner());
+    Command::new(std::env::current_exe().expect("test executable"))
+        .args(["--exact", &name, "--nocapture", "--test-threads=1"])
+        .env(CHILD, &name)
+        .output()
+        .expect("launch isolated test")
+}
+
+/// True in the parent after reporting the child's result; false in the
+/// child, which must run the original test body unchanged.
+pub(crate) fn isolate(module: &str, name: &str) -> bool {
+    if in_child(module, name) {
+        return false;
+    }
+    let output = run_child(module, name);
+    let stdout = String::from_utf8_lossy(&output.stdout);
+    let stderr = String::from_utf8_lossy(&output.stderr);
+    // Do not accept an accidentally unmatched --exact filter (zero tests).
+    assert!(
+        output.status.success() && stdout.contains("test result: ok. 1 passed"),
+        "isolated {module}::{name}: {}\n{stdout}\n{stderr}",
+        output.status
+    );
+    print!("{stdout}");
+    eprint!("{stderr}");
+    true
+}
+
+#[test]
+fn fatal_gpu_failure_is_reported_without_runtime_callback() {
+    const NAME: &str = "fatal_gpu_failure_is_reported_without_runtime_callback";
+    if in_child(module_path!(), NAME) {
+        // A callback would be the runtime's fatal dialog. It must never run.
+        crate::host::on_failure(|_| std::process::exit(77));
+        crate::host::fail(format_args!("V5 deliberate fatal GPU test failure"));
+    }
+    let output = run_child(module_path!(), NAME);
+    assert_eq!(output.status.code(), Some(101), "{output:?}");
+    let stderr = String::from_utf8_lossy(&output.stderr);
+    assert!(stderr.contains("V5 deliberate fatal GPU test failure"), "{stderr}");
+}
