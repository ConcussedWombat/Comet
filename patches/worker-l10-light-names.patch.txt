+/// Candidate name markers only: a substring of a container is not a
+/// verified entry point, or evidence that a dispatch produced an output.
+#[cfg(all(windows, feature = "diagnostics"))]
+fn l10_light_markers(bytes: &[u8]) -> impl Iterator<Item = &'static str> + '_ {
+    const MARKERS: &[&str] = &[
+        "indirect_diffuse_lighting",
+        "CS_ScreenProbes_TemporalResolve",
+        "CS_ScreenProbes_TraceRays",
+        "CS_ScreenProbes",
+        "IrradianceCache",
+        "RadianceCache",
+    ];
+    MARKERS.iter().copied().filter(move |name| {
+        bytes.windows(name.len()).any(|w| w == name.as_bytes())
+    })
+}
+
+#[cfg(all(windows, feature = "diagnostics", test))]
+#[test]
+fn l10_light_marker_candidates() {
+    let bytes = b"prefix\0CS_ScreenProbes_TemporalResolve\0suffix";
+    assert_eq!(
+        l10_light_markers(bytes).collect::<Vec<_>>(),
+        vec!["CS_ScreenProbes_TemporalResolve", "CS_ScreenProbes"]
+    );
+    assert_eq!(l10_light_markers(b"unrelated").count(), 0);
+}
