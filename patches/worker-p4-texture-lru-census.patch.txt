+/// An SRV was prepared for a draw or dispatch, not the background heap
+/// texture pass. Never interpret an old or absent stamp as proof of no use:
+/// shaders can index the heap directly, and preparation is not execution.
+#[cfg(all(windows, feature = "diagnostics"))]
+pub(crate) fn p4_note_prepared(this: This) {
+    if p4_lru_trace() && !this.is_null() {
+        let r = resource(this);
+        if fills(&r.desc) {
+            r.host.p4_prepared.fetch_max(crate::frame::current().saturating_add(1),
+                std::sync::atomic::Ordering::Relaxed);
+        }
+    }
+}

+/// Bucket order: never prepared, prepared >= 4 frames ago, prepared more
+/// recently. A concurrent/newer stamp is conservatively recent.
+#[cfg(all(windows, feature = "diagnostics"))]
+fn p4_prepared_bucket(frame: u64, stamp: u64) -> usize {
+    if stamp == 0 { 0 } else if frame.saturating_sub(stamp - 1) >= 4 { 1 } else { 2 }
+}

+/// Mutually exclusive pin buckets: Arc shared (possibly also COM shared),
+/// COM shared/unknown with sole Arc, sole Arc and sole COM ref. The latter
+/// is not proof of eviction eligibility: age and host writes also matter.
+#[cfg(all(windows, feature = "diagnostics"))]
+fn p4_pin_bucket(arcs: usize, refs: u32) -> usize {
+    if arcs != 1 { 0 } else if refs != 1 { 1 } else { 2 }
