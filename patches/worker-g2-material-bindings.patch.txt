+/// G2: diagnostic only, exact native stage hashes, never a global switch.
+#[cfg(feature = "diagnostics")]
+fn g2_material_selected(hash: &str) -> bool {
+    static HASHES: std::sync::OnceLock<Vec<String>> = std::sync::OnceLock::new();
+    HASHES.get_or_init(|| {
+        std::env::var("COMET_G2_MATERIAL_TRACE").unwrap_or_default().split(',')
+            .map(str::trim)
+            .filter(|s| s.len() == 16 && s.bytes().all(|b| b.is_ascii_hexdigit()))
+            .map(str::to_ascii_lowercase).collect()
+    }).iter().any(|s| s == hash)
+}

+#[cfg(feature = "diagnostics")]
+fn g2_record_range(base: u64, len: u64, offset: u64, bytes: u64) -> Option<u64> {
+    if offset.checked_add(bytes)? > len {
+        return None;
+    }
+    let start = base.checked_add(offset)?;
+    start.checked_add(bytes)?;
+    Some(start)
+}
+
+#[cfg(all(test, feature = "diagnostics"))]
+#[test]
+fn g2_record_range_is_bounded() {
+    assert_eq!(g2_record_range(0x1000, 128, 64, 64), Some(0x1040));
+    assert_eq!(g2_record_range(0x1000, 128, 65, 64), None);
+    assert_eq!(g2_record_range(u64::MAX - 1, 4, 0, 4), None);
+    assert_eq!(g2_record_range(0, u64::MAX, u64::MAX, 1), None);
+}

+    /// CPU recording-time binding snapshot. No mirror, upload, queue
+    /// retirement, state transition, or change to shader acceptance.
+    #[cfg(feature = "diagnostics")]
+    fn g2_material_trace(&self, hash: &str, used: &[UsedResource],
+        indexed: &[RootIndexedView], made: &[(u8, u32)], compute: bool) {
+        static CALLS: AtomicU64 = AtomicU64::new(0);
+        if !crate::capture::listed(crate::queue::presented() + 1)
+            || CALLS.fetch_add(1, Ordering::Relaxed) >= 2048 {
+            return;
+        }
+        let roots = self.roots_ref(compute);
+        tracing::info!(hash, compute, frame = crate::frame::current(),
+            pipeline = self.pipeline, targets = ?self.targets, depth = self.depth,
+            ?used, ?indexed, ?made, "G2 material stage");
+        let bind = |slot: u8, offset: u32, kind: Option<ResourceType>| {
+            let resolved = (|| {
+                let Root::Table(table) = roots.get(usize::from(slot))? else { return None };
+                let heap = self.heap_of(*table)?;
+                if heap.samplers() || (table - heap.start) & 31 != 0 || offset & 31 != 0 {
+                    return None;
+                }
+                let handle = table.checked_add(u64::from(offset))?;
+                let original = heap.index(handle)?;
+                let relocation = made.iter().find(|m| m.0 == slot);
+                let actual = match relocation {
+                    Some(&(_, base)) => base.checked_add(offset)? / 32,
+                    None => original,
+                };
+                Some((heap.host()?, handle, original, actual, relocation.is_some()))
+            })();
+            let Some((host, handle, original, actual, relocated)) = resolved else {
+                tracing::info!(hash, slot, offset, ?kind, "G2 unresolved material binding");
+                return;
+            };
+            let view = crate::descriptor::view(handle);
+            let mirror = if relocated { None } else { host.describe_slot(actual) };
+            let backing = host.resource_at(actual).map(|resource| {
+                // SAFETY: the cloned resource remains live for the query.
+                let desc = unsafe { resource.GetDesc() };
+                (resource.as_raw() as usize, desc.Format.0, desc.Width, desc.Height,
+                    desc.DepthOrArraySize, desc.MipLevels, desc.Flags.0)
+            });
+            tracing::info!(hash, slot, offset, ?kind, original, actual, relocated, ?view,
+                ?mirror, ?backing, "G2 material binding");
+        };
+        for u in used.iter().filter(|u| u.root == RootType::Table) {
+            if let Some(offset) = u.table_offset_dwords.checked_mul(4) {
+                bind(u.slot, offset, Some(u.kind));
+            }
+        }
+        for index in indexed {
+            let offset = (|| {
+                let Root::Table(table) = roots.get(usize::from(index.table_slot))? else { return None };
+                let Root::Constants(c) = roots.get(usize::from(index.constant_slot))? else { return None };
+                let constant = *c.get(index.constant_dword as usize)?;
+                let heap = self.heap_of(*table)?;
+                let base = u32::try_from(table.checked_sub(heap.start)?).ok()?;
+                let offset = index.byte_offset.wrapping_add(constant.wrapping_shl(u32::from(index.shift)));
+                let absolute = base.checked_add(offset)?;
+                (absolute & 31 == 0 && base & 31 == 0).then_some(offset)
+            })();
+            if let Some(offset) = offset {
+                bind(index.table_slot, offset, None);
+            } else {
+                tracing::info!(hash, ?index, "G2 unresolved indexed material address");
+            }
+        }
+        if !compute {
+            self.g2_ped_records(hash, used);
+        }
+    }

+    /// Candidate words only: a number that names a texture does not prove
+    /// that the shader samples it. Never write or prepare such a slot.
+    #[cfg(feature = "diagnostics")]
+    fn g2_ped_records(&self, hash: &str, used: &[UsedResource]) {
+        let roots = self.roots_ref(false);
+        let uses = |slot, dword| used.iter().any(|u| u.root == RootType::Table
+            && u.slot == slot && u.table_offset_dwords == dword);
+        if !uses(1, 56) || !uses(0, 160) {
+            return;
+        }
+        let range = |slot: u8, dword: u32| {
+            let Root::Table(table) = roots.get(usize::from(slot))? else { return None };
+            let handle = table.checked_add(u64::from(dword) * 4)?;
+            self.heap_of(handle)?;
+            let view = crate::descriptor::view(handle)?;
+            match view {
+                View::Constant(address, size) => Some((address, u64::from(size))),
+                other => crate::descriptor::host::buffer_range(&other),
+            }
+        };
+        let Some((constants, size)) = range(1, 56) else { return };
+        let Some(at) = g2_record_range(constants, size, 0x1c, 4) else { return };
+        let mut word = [0; 4];
+        if !crate::host::read_into(at as *const u8, &mut word) { return; }
+        let material = u64::from(u32::from_le_bytes(word));
+        let Some((base, len)) = range(0, 160) else { return };
+        let Some(heap) = (self.heaps[0] != 0)
+            .then(|| crate::descriptor::heap(self.heaps[0] as This)) else { return };
+        let Some(host) = heap.host() else { return };
+        tracing::info!(hash, constants, size, material, base, len, "G2 ped record candidate");
+        let probe = |label: &str, base: u64, len: u64, offset: u64, bytes: usize| {
+            let Some(addr) = g2_record_range(base, len, offset, bytes as u64) else { return };
+            let mut data = [0u8; 512];
+            if !crate::host::read_into(addr as *const u8, &mut data[..bytes]) {
+                tracing::info!(hash, label, addr, bytes, "G2 ped record unreadable");
+                return;
+            }
+            for (i, w) in data[..bytes].chunks_exact(4).enumerate() {
+                let slot = u32::from_le_bytes([w[0], w[1], w[2], w[3]]);
+                if slot == 0 { continue; }
+                let Some(view) = heap.handle(slot).and_then(crate::descriptor::view) else { continue };
+                if !crate::descriptor::host::is_texture_view(&view) { continue; }
+                tracing::info!(hash, label, record_offset = offset, word_offset = i * 4,
+                    slot, ?view, mirror = ?host.describe_slot(slot),
+                    backed = host.slot_has_resource(slot), "G2 ped texture candidate");
+            }
+        };
+        probe("M", base, len, material, 64);
+        if let Some(offset) = material.checked_add(1936) {
+            probe("M+1936", base, len, offset, 512);
+        }
+        if uses(0, 688)
+            && let Some(slot) = used.iter().find(|u| u.root == RootType::Constants).map(|u| u.slot)
+            && let Some(Root::Constants(words)) = roots.get(usize::from(slot))
+            && let Some(&offset) = words.get(1)
+            && let Some((base, len)) = range(0, 688)
+        {
+            probe("G", base, len, u64::from(offset), 64);
+        }
+    }
