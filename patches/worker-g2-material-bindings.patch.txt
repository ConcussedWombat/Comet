+/// Exact native stage hashes only; never enable material repair globally.
+fn g2_selected(hash: &str, repair: bool) -> bool {
+    static TRACE: std::sync::OnceLock<Vec<String>> = std::sync::OnceLock::new();
+    static REPAIR: std::sync::OnceLock<Vec<String>> = std::sync::OnceLock::new();
+    let (cache, name) = if repair {
+        (&REPAIR, "COMET_G2_INDEXED_SRV")
+    } else {
+        (&TRACE, "COMET_G2_MATERIAL_TRACE")
+    };
+    cache.get_or_init(|| {
+        std::env::var(name).unwrap_or_default().split(',')
+            .map(str::trim)
+            .filter(|s| s.len() == 16 && s.bytes().all(|b| b.is_ascii_hexdigit()))
+            .map(str::to_ascii_lowercase).collect()
+    }).iter().any(|s| s == hash)

+/// Only an aligned, non-wrapping address in the original table is proven
+/// by this probe. Other cases are left to the existing path.
+fn g2_indexed_byte_offset(index: &RootIndexedView, constant: u32, base: u32) -> Option<u32> {
+    let offset = index.byte_offset.wrapping_add(constant.wrapping_shl(u32::from(index.shift)));
+    let absolute = base.checked_add(offset)?;
+    (absolute % 32 == 0 && base % 32 == 0).then_some(offset)
+}
+
+#[test]
+fn g2_indexed_byte_offset_requires_aligned_nonwrapping_base() {
+    let mut index = RootIndexedView {
+        table_slot: 4, constant_slot: 2, constant_dword: 0, byte_offset: 64, shift: 5,
+    };
+    assert_eq!(g2_indexed_byte_offset(&index, 7, 32), Some(288));
+    assert_eq!(g2_indexed_byte_offset(&index, u32::MAX, 32), Some(32));
+    assert_eq!(g2_indexed_byte_offset(&index, 7, u32::MAX - 31), None);
+    assert_eq!(g2_indexed_byte_offset(&index, 7, 4), None);
+    index.byte_offset = 4;
+    assert_eq!(g2_indexed_byte_offset(&index, 0, 0), None);
+}
