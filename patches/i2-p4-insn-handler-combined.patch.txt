+fn p4_insn_span(what: ConsoleOnly) -> Option<P4InsnSpan> {
+    use windows_sys::Win32::Foundation::{GetLastError, SetLastError};
+    static ON: std::sync::OnceLock<bool> = std::sync::OnceLock::new();
+    if ON.get() == Some(&false) {
+        return None;
+    }
+    // SAFETY: last-error belongs to this thread.
+    let last = unsafe { GetLastError() };
+    let on = *ON.get_or_init(|| std::env::var_os("COMET_P4_INSN_PROFILE").is_some_and(|v| v == "1"));
+    let span = on.then(|| P4InsnSpan { kind: instruction_kind(what), start: std::time::Instant::now() });
+    unsafe { SetLastError(last) };
+    span
+}

+/// Each successful emulation used to modify a single global cache line.
+/// Keep exact per-shard counts; report their sum periodically instead of
+/// bouncing one counter among all title workers on every illegal instruction.
+const P4_SHARDS: usize = 64;
+const P4_POLL: u64 = 256;
+
+#[repr(align(64))]
+struct P4Count(AtomicU64);
+
+struct P4Counters {
+    shards: [P4Count; P4_SHARDS],
+    next: AtomicU64,
+    report: AtomicU64,
+}
+
+impl P4Counters {
+    const fn new() -> Self {
+        Self {
+            shards: [const { P4Count(AtomicU64::new(0)) }; P4_SHARDS],
+            next: AtomicU64::new(0),
+            report: AtomicU64::new(1 << 16),
+        }
+    }
+
+    fn assign(&self) -> usize {
+        self.next.fetch_add(1, Relaxed) as usize % P4_SHARDS
+    }
+
+    fn bump(&self, shard: usize) -> u64 {
+        self.shards[shard].0.fetch_add(1, Relaxed).wrapping_add(1)
+    }
+
+    fn total(&self) -> u64 {
+        self.shards.iter().map(|shard| shard.0.load(Relaxed)).sum()
+    }
+
+    /// At most one reporting thread claims each threshold. The cumulative
+    /// sample is not transactional across threads; each completed increment
+    /// stays in its shard, including when its owner exits.
+    fn report_at(&self, count: u64) -> Option<u64> {
+        let threshold = self.report.load(Relaxed);
+        if count < threshold {
+            return None;
+        }
+        let next = count.checked_next_power_of_two()
+            .and_then(|n| if n == count { n.checked_mul(2) } else { Some(n) })
+            .unwrap_or(u64::MAX);
+        if next == threshold {
+            return None;
+        }
+        self.report.compare_exchange(threshold, next, Relaxed, Relaxed).ok().map(|_| count)
+    }
+}
