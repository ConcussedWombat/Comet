+fn record_gathers_with(c: &Clamp, l: &ID3D12GraphicsCommandList, sets: &[GatherSet], budget: u64) -> bool {
+    if sets.is_empty() || !budget.is_multiple_of(4) {
+        return false;
+    }
+    for set in sets {
+        // SAFETY: resources owned by the recorded list.
+        let (src, dst) = unsafe { (set.src.GetGPUVirtualAddress(), set.scratch.GetGPUVirtualAddress()) };
+        if set.records == 0 || set.records > GATHER_MOST
+            || gatherable(src, 0, (set.kind, set.stride)).is_none()
+            || !dst.is_multiple_of(4) || !set.table_gpu.is_multiple_of(4) {
+            return false;
+        }
+        let constants = [set.kind, set.stride, set.records, 0, 4, 0];
+        // SAFETY: open dedicated prelude, live table, budget and scratch.
+        unsafe {
+            l.ResourceBarrier(&[uav_barrier(), transition(&set.scratch,
+                D3D12_RESOURCE_STATE_COMMON, D3D12_RESOURCE_STATE_UNORDERED_ACCESS)]);

+            // clamps; transition publishes scratch to the consumer list.
