+pub(crate) fn record_gathers(l: &ID3D12GraphicsCommandList, sets: &[GatherSet], budget: u64) -> bool {
+    let Some(c) = gather_clamp() else { return false };
+    if sets.is_empty() || !budget.is_multiple_of(4) {
+        return false;
+    }
+    for set in sets {

+        let (src, dst) = unsafe { (set.src.GetGPUVirtualAddress(), set.scratch.GetGPUVirtualAddress()) };
