+// A made-key hit is not sufficient by itself: releasing a resource
+// invalidates its generation without clearing every slot that names it.
+// The caller holds the heap's made read guard while resolving the view.
+fn p2_mirror_hit(made: &[u64], index: u32, tag: u32, i: u32, generation: u32) -> Option<View> {
+    let key = u64::from(generation) << 32 | u64::from(i);
+    if tag != VIEW_TAG || key == 0 || made.get(index as usize).copied() != Some(key) {
+        return None;
+    }
+    view_at(i, generation)
+}

+#[test]
+fn p2_shared_mirror_checks_generation_and_key() {
+    // Only registry metadata is used, never dereference the fake resource.
+    let resource = 0x52a0_0000usize;
+    let view = View::Shader(resource, None);
+    let [i, generation] = super::words_for(view.clone());
+    let made = [u64::from(generation) << 32 | u64::from(i), 0];
+    assert_eq!(p2_mirror_hit(&made, 0, VIEW_TAG, i, generation), Some(view.clone()));
+    assert_eq!(p2_mirror_hit(&made, 0, 0, i, generation), None);
+    assert_eq!(p2_mirror_hit(&made, 1, VIEW_TAG, i, generation), None);
+    assert_eq!(p2_mirror_hit(&made, 2, VIEW_TAG, i, generation), None);
+    super::forget(resource);
+    // The heap's cached key is still equal, but the registry invalidated it.
+    assert_eq!(p2_mirror_hit(&made, 0, VIEW_TAG, i, generation), None);
+    let [new_i, new_generation] = super::words_for(view);
+    assert_ne!([i, generation], [new_i, new_generation]);
+    assert_eq!(p2_mirror_hit(&made, 0, VIEW_TAG, new_i, new_generation), None);
+    super::forget(resource);
+}
