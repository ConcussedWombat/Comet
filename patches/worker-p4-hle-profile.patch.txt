+//! Opt-in cumulative HLE timing; completed calls, inclusive wall time (nested calls overlap).
+use std::sync::OnceLock;
+use std::sync::atomic::{AtomicU64, Ordering::Relaxed};
+use std::time::Instant;
+use windows_sys::Win32::Foundation::{GetLastError, SetLastError};
+
+#[repr(usize)]
+#[derive(Clone, Copy)]
+pub(crate) enum Kind {
+    VirtualAlloc,
+    VirtualQuery,
+    Alloc,
+    Free,
+    AllocDefault,
+    FreeDefault,
+    PhysicalAlloc,
+    PhysicalFree,
+    PhysicalMap,
+    PhysicalScatter,
+    TaskDispatch,
+    TaskSubmit,
+    AsyncStatus,
+    AsyncResult,
+    ReadFile,
+    Fread,
+    CreateFileW,
+    CreateFileA,
+    StoragePoll,
+    DebugStringA,
+    DebugStringW,
+}
+const N: usize = 21;
+const NAMES: [&str; N] = [
+    "XMemVirtualAlloc",
+    "XMemVirtualQuery",
+    "XMemAlloc",
+    "XMemFree",
+    "XMemAllocDefault",
+    "XMemFreeDefault",
+    "XMemAllocatePhysicalPages",
+    "XMemFreePhysicalPages",
+    "XMemMapPhysicalPages",
+    "XMemMapPhysicalPagesScatter",
+    "XTaskQueueDispatch",
+    "XTaskQueueSubmitDelayedCallback",
+    "XAsyncGetStatus",
+    "XAsyncGetResult",
+    "ReadFile",
+    "fread",
+    "CreateFileW",
+    "CreateFileA",
+    "is_complete",
+    "OutputDebugStringA",
+    "OutputDebugStringW",
+];
+
+struct Stat {
+    calls: AtomicU64,
+    nanos: AtomicU64,
+}
+static STATS: [Stat; N] = [const { Stat { calls: AtomicU64::new(0), nanos: AtomicU64::new(0) } }; N];
+static EPOCH: OnceLock<Instant> = OnceLock::new();
+static NEXT_MS: AtomicU64 = AtomicU64::new(1000);
+
+fn enabled() -> bool {
+    static ON: OnceLock<bool> = OnceLock::new();
+    if let Some(&on) = ON.get() {
+        return on;
+    }
+    // SAFETY: last-error belongs to this thread.
+    let last = unsafe { GetLastError() };
+    let on = *ON.get_or_init(|| std::env::var_os("COMET_P4_HLE_PROFILE").is_some_and(|v| v == "1"));
+    unsafe { SetLastError(last) };
+    on
+}
+
+pub(crate) struct Span {
+    kind: Kind,
+    start: Instant,
+}
+
+pub(crate) fn start(kind: Kind) -> Option<Span> {
+    if !enabled() {
+        return None;
+    }
+    // SAFETY: last-error belongs to this thread.
+    let last = unsafe { GetLastError() };
+    let start = Instant::now();
+    EPOCH.get_or_init(|| start);
+    unsafe { SetLastError(last) };
+    Some(Span { kind, start })
+}
+
+impl Drop for Span {
+    fn drop(&mut self) {
+        // Preserve the wrapped API's resulting last-error, including failure.
+        // SAFETY: last-error belongs to this thread.
+        let last = unsafe { GetLastError() };
+        let stat = &STATS[self.kind as usize];
+        stat.nanos.fetch_add(self.start.elapsed().as_nanos() as u64, Relaxed);
+        stat.calls.fetch_add(1, Relaxed);
+        let ms = EPOCH.get_or_init(Instant::now).elapsed().as_millis() as u64;
+        let next = NEXT_MS.load(Relaxed);
+        if ms >= next && NEXT_MS.compare_exchange(next, ms.saturating_add(1000), Relaxed, Relaxed).is_ok() {
+            let snapshot: Vec<_> = STATS.iter().zip(NAMES).filter_map(|(stat, name)| {
+                let calls = stat.calls.load(Relaxed);
+                (calls != 0).then(|| (name, calls, stat.nanos.load(Relaxed) / 1000))
+            }).collect();
+            tracing::info!(elapsed_ms = ms, ?snapshot, "P4 HLE cumulative (API, completed calls, inclusive microseconds)");
+        }
+        unsafe { SetLastError(last) };
+    }
+}
