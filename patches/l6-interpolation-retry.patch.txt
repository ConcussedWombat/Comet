+/// Scope the existing independent-half semantics to a single retry.
+/// Restore even if lowering unwinds, and never mutate process settings.
+pub(crate) fn with_independent_interp<T>(f: impl FnOnce() -> T) -> T {
+    struct Restore(bool);
+    impl Drop for Restore {
+        fn drop(&mut self) {
+            L6_RETRY_INDEPENDENT.set(self.0);
+        }
+    }
+    let _restore = Restore(L6_RETRY_INDEPENDENT.replace(true));
+    f()
+}
