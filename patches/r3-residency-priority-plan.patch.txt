+//! Stage 4c: deterministic pageable residency policy, with real DXGI
+//! budget observation. No calls to Evict/MakeResident/SetResidencyPriority
+//! until the completed feedback and submission ownership join is wired.
+use std::sync::OnceLock;
+use std::sync::atomic::{AtomicBool, AtomicU64, Ordering::Relaxed};
+
+pub(super) fn enabled() -> bool {
+    static ON: OnceLock<bool> = OnceLock::new();
+    *ON.get_or_init(|| std::env::var_os("COMET_R3_RESIDENCY_POLICY").is_some_and(|v| v == "1"))
+}
+
+#[derive(Debug, Clone, Copy, PartialEq, Eq)]
+enum Action {
+    Pin,
+    Normal,
+    Low,
+    Evict,
+    MakeResident,
+}
+
+#[derive(Debug, Clone, Copy)]
+struct Pageable {
+    /// All occupants observed with exact generation and full coverage.
+    complete: bool,
+    /// Maximum last-use timestamp of all live occupants, monotonic ms.
+    last_use: u64,
+    /// The greatest graphics/copy use fence; false also covers unsubmitted
+    /// recorded lists that can still name this pageable.
+    idle: bool,
+    /// Positive upcoming use, including explicit table/copy accesses.
+    needed: bool,
+    resident: bool,
+    bytes: u64,
+}
+
+fn pressure(previous: bool, budget: u64, used: u64) -> bool {
+    if budget == 0 {
+        return false;
+    }
+    let high = budget.saturating_sub(budget / 10);
+    let low = budget.saturating_sub(budget / 5);
+    if previous { used > low } else { used >= high }
+}
+
+fn choose(p: Pageable, now: u64, cold_ms: u64, pressured: bool, nonlocal_headroom: u64) -> Action {
+    // Any unproven consumer is a pin, not a negative use observation.
+    if p.needed || !p.complete {
+        return if p.resident { Action::Normal } else { Action::MakeResident };
+    }
+    if !p.idle || p.last_use > now {
+        return if p.resident { Action::Pin } else { Action::MakeResident };
+    }
+    if !p.resident {
+        return Action::Pin;
+    }
+    if now - p.last_use < cold_ms {
+        return Action::Normal;
+    }
+    if pressured && p.bytes != 0 && p.bytes <= nonlocal_headroom {
+        Action::Evict
+    } else {
+        Action::Low
+    }
+}
+
+pub(super) fn observe(memory: Option<[u64; 4]>) {
+    static PRESSURE: AtomicBool = AtomicBool::new(false);
+    static COUNT: AtomicU64 = AtomicU64::new(0);
+    let Some([budget, used, nonlocal, nonlocal_used]) = memory else { return };
+    let pressured = pressure(PRESSURE.load(Relaxed), budget, used);
+    PRESSURE.store(pressured, Relaxed);
+    let nonlocal_headroom = nonlocal.saturating_sub(nonlocal / 10).saturating_sub(nonlocal_used);
+    // Until a generation-complete join exists, every pageable is unknown.
+    // Do not actually demote/evict or alter admission.
+    let action = choose(Pageable {
+        complete: false, last_use: 0, idle: false, needed: false,
+        resident: true, bytes: 0,
+    }, crate::host::now_ms(), 5000, pressured, nonlocal_headroom);
+    let n = COUNT.fetch_add(1, Relaxed) + 1;
+    if n.is_power_of_two() || n.is_multiple_of(4096) {
+        tracing::info!(n, budget, used, nonlocal, nonlocal_used, pressured,
+            nonlocal_headroom, ?action, "R3 residency policy (feedback unavailable; no actions)");
+    }
+}

+    #[test]
+    fn r3_priority_requires_complete_fences_and_budget() {
+        let cold = Pageable { complete: true, last_use: 100, idle: true,
+            needed: false, resident: true, bytes: 32 };
+        assert_eq!(choose(cold, 6000, 5000, true, 32), Action::Evict);
+        assert_eq!(choose(cold, 6000, 5000, true, 31), Action::Low);
+        assert_eq!(choose(cold, 6000, 5000, false, 32), Action::Low);
+        assert_eq!(choose(Pageable { complete: false, ..cold }, 6000, 5000, true, 32), Action::Normal);
+        assert_eq!(choose(Pageable { idle: false, ..cold }, 6000, 5000, true, 32), Action::Pin);
+        assert_eq!(choose(Pageable { last_use: 5999, ..cold }, 6000, 5000, true, 32), Action::Normal);
+        assert_eq!(choose(Pageable { needed: true, resident: false, ..cold }, 6000, 5000, true, 32), Action::MakeResident);
+        assert_eq!(choose(Pageable { complete: false, resident: false, ..cold }, 6000, 5000, true, 32), Action::MakeResident);
+        assert_eq!(choose(Pageable { resident: false, ..cold }, 6000, 5000, true, 32), Action::Pin);
+    }

+    /// Real API smoke test on a private heap, never a title allocation.
+    /// Opt-in so ordinary CI does not change the test device's residency.
+    #[test]
+    fn r3_priority_driver_round_trip() {
+        if !enabled() {
+            return;
+        }
+        use windows::Win32::Graphics::Direct3D12::*;
+        use windows::core::Interface;
+        let Some(host) = crate::host::host() else { return };
+        let Ok(device1) = host.device.cast::<ID3D12Device1>() else { return };
+        let desc = D3D12_HEAP_DESC {
+            SizeInBytes: 64 << 10,
+            Properties: D3D12_HEAP_PROPERTIES {
+                Type: D3D12_HEAP_TYPE_DEFAULT,
+                ..Default::default()
+            },
+            Alignment: 64 << 10,
+            Flags: D3D12_HEAP_FLAG_ALLOW_ONLY_NON_RT_DS_TEXTURES,
+        };
+        let mut heap: Option<ID3D12Heap> = None;
+        // SAFETY: private heap, no resource/list/descriptor can reference it.
+        unsafe { host.device.CreateHeap(&desc, &mut heap) }.unwrap();
+        let heap = heap.unwrap();
+        let objects = [Some(heap.cast::<ID3D12Pageable>().unwrap())];
+        let low = [D3D12_RESIDENCY_PRIORITY_LOW];
+        let normal = [D3D12_RESIDENCY_PRIORITY_NORMAL];
+        // SAFETY: each array has one live entry, the heap is initially
+        // resident and no GPU work can name it. Exactly one Evict/MakeResident.
+        unsafe {
+            device1.SetResidencyPriority(1, objects.as_ptr(), low.as_ptr()).unwrap();
+            host.device.Evict(&objects).unwrap();
+            host.device.MakeResident(&objects).unwrap();
+            device1.SetResidencyPriority(1, objects.as_ptr(), normal.as_ptr()).unwrap();
+        }
+        tracing::info!("R3 private residency API round trip succeeded");
+    }
