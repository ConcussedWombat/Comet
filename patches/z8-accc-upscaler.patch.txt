+/// Adapt the leader's FSPX resource-record layout to Comet's NativeProgram.
+/// A code section is accepted only when metadata proves its exact extent.
+/// Do not infer a root parameter or a wave mode from a shader name.
+fn z8_fspx(part: &[u8]) -> Result<NativeProgram, ShaderError> {
+    use crate::dxbc::u16_at;
+    if !part.starts_with(b"FSPX") || u32_at(part, 4)? != 1 {
+        return Err(Malformed("Z8 unknown FSPX version"));
+    }
+    let count = u32_at(part, 8)? as usize;
+    if !(3..=MAX_XDAT_ENTRIES).contains(&count)
+        || u32_at(part, 12)? as usize != part.len()
+        || part.len() > MAX_UNPACKED as usize
+    {
+        return Err(Malformed("Z8 invalid FSPX table size"));
+    }
+    let table_end = 16 + count * 16;
+    let mut found = [None; 3];
+    for i in 0..count {
+        let at = 16 + 16 * i;
+        let kind = u32_at(part, at)?;
+        let offset = u32_at(part, at + 4)?;
+        let size = u32_at(part, at + 8)?;
+        if (offset as usize) < table_end || u32_at(part, at + 12)? != 0 {
+            return Err(Malformed("Z8 invalid FSPX entry"));
+        }
+        let section = bytes(part, offset.into(), size.into())
+            .ok_or(Malformed("Z8 FSPX entry out of bounds"))?;
+        let index = match kind {
+            0 => 0,
+            6 => 1,
+            7 => 2,
+            _ => continue,
+        };
+        if found[index].replace(section).is_some() {
+            return Err(Malformed("Z8 duplicate FSPX entry"));
+        }
+    }
+    let [Some(packet), Some(code_chunk), Some(meta)] = found else {
+        return Err(Malformed("Z8 missing FSPX entry"));
+    };
+    if packet.len() != 0x1c || meta.len() < 0x18
+        || u32_at(meta, 0)? != 0x403
+        || u32_at(meta, 4)? as usize != meta.len() - 0x14
+        || meta[20] != 0x40
+    {
+        return Err(Malformed("Z8 unsupported FSPX metadata header"));
+    }
+    let code_len = (u32_at(meta, 8)? as usize)
+        .checked_mul(4)
+        .filter(|&n| n != 0)
+        .ok_or(Malformed("Z8 invalid FSPX code length"))?;
+    // The reference fork uses the raw code entry. Comet's L3 traces also
+    // recognize a 0x24-byte code prefix; only choose the form matching the
+    // declared instruction extent, never strip bytes speculatively.
+    let code = if code_chunk.len() == code_len {
+        code_chunk
+    } else if code_len.checked_add(0x24) == Some(code_chunk.len()) {
+        &code_chunk[0x24..]
+    } else {
+        return Err(Malformed("Z8 FSPX code size disagrees"));
+    };
+    let threads = [
+        u32::from(u16_at(packet, 0xc)?),
+        u32::from(u16_at(packet, 0xe)?),
+        u32::from(u16_at(packet, 0x10)?),
+    ];
+    if threads.iter().any(|&n| !(1..=1024).contains(&n))
+        || threads.iter().product::<u32>() > 1024
+    {
+        return Err(Malformed("Z8 invalid FSPX thread group"));
+    }
+    let wave64 = u16_at(packet, 0x12)? & 0x10 == 0;
+    let vgprs = ((u32_at(packet, 0x14)? & 0x3f) + 1)
+        * if wave64 { 4 } else { 8 };
+    if vgprs > 256 {
+        return Err(Malformed("Z8 invalid FSPX VGPR allocation"));
+    }
+    let io = ComputeIo { rsrc2: u32_at(packet, 0x18)?, threads };
+    let mut layout: Vec<RootParam> = Vec::new();
+    let mut used = Vec::new();
+    let mut at = 0x18;
+    let mut ended = false;
+    while at < meta.len() {
+        let tag = u32_at(meta, at)?;
+        if tag == 0x43 {
+            // The shader-id record is not a resource descriptor.
+            if meta.len() - at < 20 {
+                return Err(Malformed("Z8 truncated FSPX hash record"));
+            }
+            ended = true;
+            break;
+        }
+        let record = meta.get(at..at + 32)
+            .ok_or(Malformed("Z8 truncated FSPX record"))?;
+        at += 32;
+        if tag == 0x42 {
+            continue; // declaration only; do not invent a binding
+        }
+        if tag != 0x41 {
+            return Err(Malformed("Z8 unknown FSPX record"));
+        }
+        let kind = match record[4] {
+            0 => ResourceType::Srv,
+            1 => ResourceType::Sampler,
+            2 => ResourceType::Cbv,
+            3 => ResourceType::Uav,
+            _ => return Err(Malformed("Z8 unknown FSPX resource type")),
+        };
+        let root = RootType::from_raw(u32::from(record[5]));
+        let u = UsedResource {
+            kind,
+            root,
+            table_offset_dwords: le32(record, 8),
+            lower_bound: le32(record, 12),
+            // The 0x41 dimension word is NOT an element stride.
+            stride: 0,
+            dwords: record[20],
+            user_data: record[21],
+            slot: record[22],
+        };
+        let sgprs = match root {
+            RootType::Table => 1,
+            RootType::Descriptor => 2,
+            RootType::ImmediateSampler => 0,
+            _ => return Err(Malformed("Z8 unsupported FSPX root type")),
+        };
+        if sgprs != 0 {
+            let end = u32::from(u.user_data) + sgprs;
+            if end > u32::from(io.user_sgprs()) {
+                return Err(Malformed("Z8 FSPX root outside user SGPRs"));
+            }
+            if let Some(previous) = layout.iter().find(|p| p.slot == u.slot) {
+                if previous.root != root || previous.user_data != u.user_data
+                    || previous.sgprs != sgprs
+                {
+                    return Err(Malformed("Z8 conflicting FSPX root slot"));
+                }
+            } else {
+                // A table is based at the root pointer, not at its first
+                // used descriptor. Keep per-resource offsets in used[].
+                layout.push(RootParam {
+                    root,
+                    table_offset_dwords: 0,
+                    sgprs,
+                    base_register: u.lower_bound,
+                    user_data: u.user_data,
+                    slot: u.slot,
+                });
+            }
+        }
+        used.push(u);
+    }
+    if !ended || used.len() != usize::from(meta[22]) {
+        return Err(Malformed("Z8 incomplete FSPX resource records"));
+    }
+    Ok(NativeProgram {
+        stage: HwStage::Cs,
+        wave64,
+        vgprs: vgprs as u16,
+        layout,
+        used,
+        io: StageIo::Cs(io),
+        code: code.to_vec(),
+    })
+}
