+fn l1_parse_samples(value: &str) -> Vec<(u32, u32, u32, u32)> {
+    value.split(';').filter_map(|item| {
+        let (size, xy) = item.trim().split_once(':')?;
+        let (w, h) = size.split_once('x')?;
+        let (x, y) = xy.split_once(',')?;
+        let (w, h, x, y): (u32, u32, u32, u32) =
+            (w.parse().ok()?, h.parse().ok()?, x.parse().ok()?, y.parse().ok()?);
+        (x < w && y < h).then_some((w, h, x, y))
+    }).take(32).collect()

+/// Locate a single uncompressed texel without including row padding or
+/// exceeding the actual allocation (last-row padding can be absent).
+fn l1_sample_range(
+    f: &D3D12_PLACED_SUBRESOURCE_FOOTPRINT,
+    row_bytes: u64,
+    row_count: u32,
+    allocation: u64,
+    x: u32,
+    y: u32,
+) -> Option<std::ops::Range<usize>> {
+    let p = f.Footprint;
+    if p.Width == 0 || p.Depth != 1 || row_count != p.Height
+        || x >= p.Width || y >= p.Height
+        || row_bytes > u64::from(p.RowPitch)
+        || row_bytes % u64::from(p.Width) != 0
+    {
+        return None;
+    }
+    let n = row_bytes / u64::from(p.Width);
+    if !matches!(n, 1 | 2 | 4 | 8 | 12 | 16) {
+        return None;
+    }
+    let start = f.Offset.checked_add(u64::from(y).checked_mul(u64::from(p.RowPitch))?)?
+        .checked_add(u64::from(x).checked_mul(n)?)?;
+    let end = start.checked_add(n)?;
+    if end > allocation {
+        return None;
+    }
+    Some(usize::try_from(start).ok()?..usize::try_from(end).ok()?)
+}

+/// Read exact bytes from the already completed capture readback, including
+/// alpha and stencil. No new GPU commands; end-of-frame values only.
+/// COMET_L1_TARGET_SAMPLES=1920x1080:640,480;1920x1080:700,480
+#[allow(clippy::too_many_arguments)]
+fn l1_log_target_samples(
+    device: &ID3D12Device,
+    desc: &D3D12_RESOURCE_DESC,
+    buffer: &ID3D12Resource,
+    footprint: &D3D12_PLACED_SUBRESOURCE_FOOTPRINT,
+    frame: u64,
+    resource: crate::object::This,
+    sub: u32,
+) {
+    static SAMPLES: std::sync::OnceLock<Vec<(u32, u32, u32, u32)>> = std::sync::OnceLock::new();
+    let samples = SAMPLES.get_or_init(|| {
+        l1_parse_samples(&std::env::var("COMET_L1_TARGET_SAMPLES").unwrap_or_default())
+    });
+    let f = footprint.Footprint;
+    if !samples.iter().any(|&(w, h, _, _)| (w, h) == (f.Width, f.Height)) {
+        return;
+    }
+    // Only ordinary uncompressed formats or the explicitly captured
+    // depth/stencil planes. Do not infer texel sizes for BC/planar data.
+    if !matches!(desc.Format.0, 1..=65 | 87..=93) {
+        return;
+    }
+    let mut row_bytes = 0;
+    let mut row_count = 0;
+    let mut actual = D3D12_PLACED_SUBRESOURCE_FOOTPRINT::default();
+    // SAFETY: query one subresource, same as read_back_of; all outputs valid.
+    unsafe {
+        device.GetCopyableFootprints(
+            desc, sub, 1, 0, Some(&mut actual),
+            Some(&mut row_count), Some(&mut row_bytes), None,
+        );
+    }
+    if actual.Offset != footprint.Offset || actual.Footprint.RowPitch != f.RowPitch
+        || actual.Footprint.Format != f.Format || actual.Footprint.Width != f.Width
+        || actual.Footprint.Height != f.Height
+    {
+        return;
+    }
+    // SAFETY: a live readback buffer, its copy has completed.
+    let allocation = unsafe { buffer.GetDesc() }.Width;
+    let mut data = std::ptr::null_mut();
+    // SAFETY: the capture list finished; each sampled range is bounds checked.
+    unsafe {
+        if buffer.Map(0, None, Some(&mut data)).is_err() {
+            return;
+        }
+        for &(w, h, x, y) in samples {
+            if (w, h) != (f.Width, f.Height) {
+                continue;
+            }
+            let Some(range) = l1_sample_range(footprint, row_bytes, row_count, allocation, x, y) else {
+                continue;
+            };
+            let bytes = std::slice::from_raw_parts(data.cast::<u8>().add(range.start), range.len());
+            tracing::info!(
+                frame, ?resource, subresource = sub, resource_format = desc.Format.0,
+                footprint_format = f.Format.0, width = w, height = h, x, y,
+                raw = ?bytes, all_bytes_zero = bytes.iter().all(|&b| b == 0),
+                "L1 TARGET SAMPLE end-of-frame"
+            );
+        }
+        buffer.Unmap(0, None);
+    }
+}
