+    use std::collections::HashMap;
+    use std::sync::{Mutex, OnceLock};
+    // Exclude size, address and resource lifetime: repeated allocation
+    // queries do not grow the map per texture.
+    type Key = (u32, u32, u32, u32, u32, &'static str);
+    static COUNTS: OnceLock<Mutex<HashMap<Key, u64>>> = OnceLock::new();
+    let class = r3_layout_class(desc, result);
+    let key = (
+        desc.format,
+        desc.layout,
+        desc.dimension,
+        desc.flags,
+        desc.sample_count,
+        class,
+    );
+    let count = {
+        let mut counts = COUNTS.get_or_init(|| Mutex::new(HashMap::new()))
+            .lock().unwrap_or_else(|e| e.into_inner());
+        // Bound diagnostic-only memory use even for malformed descriptions.
+        if counts.len() >= 4096 && !counts.contains_key(&key) {
+            return;
+        }
+        let count = counts.entry(key).or_default();
+        *count = count.saturating_add(1);
+        *count
