+fn float_class(bits: u32, mantissa: u32, exponent: u32, signed: bool) -> usize {
+    let mask = (1u32 << exponent) - 1;
+    if (bits >> mantissa) & mask != mask {
+        0
+    } else if bits & ((1u32 << mantissa) - 1) != 0 {
+        3
+    } else if signed && bits >> (mantissa + exponent) != 0 {
+        2
+    } else {
+        1
+    }
+}

+// do not use half()/small_float(), which intentionally collapse NaN into

+fn float_classes(format: u32, w: usize, h: usize, pitch: usize, rows: &[u8]) -> Option<FloatClasses> {
+    let (channels, bytes, packed) = match format {
+        2 => (4, 4, false),   // R32G32B32A32_FLOAT
+        6 => (3, 4, false),   // R32G32B32_FLOAT
+        10 => (4, 2, false),  // R16G16B16A16_FLOAT
+        16 => (2, 4, false),  // R32G32_FLOAT
+        26 => (3, 0, true),   // R11G11B10_FLOAT
+        34 => (2, 2, false),  // R16G16_FLOAT
+        41 => (1, 4, false),  // R32_FLOAT
+        54 => (1, 2, false),  // R16_FLOAT
+        _ => return None,
+    };
+    let pixel = if packed { 4 } else { channels * bytes };
+    let row = w.checked_mul(pixel)?;
+    if row > pitch || h.checked_mul(pitch)? > rows.len() {
+        return None;
+    }
+    let mut stats = FloatClasses::default();
+    for y in 0..h {
+        for (x, p) in rows[y * pitch..y * pitch + row].chunks_exact(pixel).enumerate() {
+            for c in 0..channels {
+                let (bits, class) = if packed {
+                    let word = u32::from_le_bytes(p.try_into().ok()?);
+                    let (shift, mantissa) = [(0, 6), (11, 6), (22, 5)][c];
+                    let bits = (word >> shift) & ((1u32 << (mantissa + 5)) - 1);
+                    (bits, float_class(bits, mantissa, 5, false))
+                } else if bytes == 2 {
+                    let bits = u32::from(u16::from_le_bytes(p[c * 2..c * 2 + 2].try_into().ok()?));
+                    (bits, float_class(bits, 10, 5, true))
+                } else {

+                    (bits, float_class(bits, 23, 8, true))
+                };

+                if class != 0 && stats.first_invalid.is_none() {
+                    stats.first_invalid = Some((x, y, c, bits));
+                }
+            }
+        }
+    }
