+impl Releasable {
+    /// A recorded copy owns a second Arc until its command list is reset.
+    fn release_if_unleased(&mut self, uploaded: bool) -> bool {
+        if !uploaded || !self.upload.as_ref().is_some_and(|u| Arc::strong_count(u) == 1) {
+            return false;
+        }
+        self.upload = None;
+        true
+    }
+
+    fn restore(&mut self, bytes: u64) -> Option<(Arc<StagingUpload>, bool)> {
+        if let Some(upload) = &self.upload {
+            return Some((upload.clone(), false));
+        }
+        let upload = StagingUpload::new(bytes)?;
+        // SAFETY: new mapped upload of this length; the backing mutex
+        // protects the CPU snapshot. No GPU list names this allocation.
+        unsafe { std::ptr::copy_nonoverlapping(self.linear.as_ptr(), upload.data, self.linear.len()) };
+        self.upload = Some(upload.clone());
+        Some((upload, true))
+    }
+}

+// B4 first cut: only the existing resident-BC path (zero initialized and
+// explicit preservation of missing mips). Bound the ordinary CPU cache on
+// the 32 GB reference machine. Allocation failure keeps the old path.
+const CPU_MIRROR_MAX: u64 = 2 << 30;
+static CPU_MIRROR: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
+static LOGICAL_BYTES: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
+static RELEASED: [std::sync::atomic::AtomicU64; 4] = [const { std::sync::atomic::AtomicU64::new(0) }; 4];

+fn releasable_upload(upload: Arc<StagingUpload>) -> Backing {
+    use std::sync::atomic::Ordering::Relaxed;
+    let Ok(size) = usize::try_from(upload.bytes) else { return Backing::Persistent(upload) };
+    // Called under STAGING's mutex; concurrent allocation cannot race the cap.
+    if upload.bytes > CPU_MIRROR_MAX.saturating_sub(CPU_MIRROR.load(Relaxed)) {
+        return Backing::Persistent(upload);
+    }
+    let mut linear = Vec::new();
+    if linear.try_reserve_exact(size).is_err() {
+        return Backing::Persistent(upload);
+    }
+    linear.resize(size, 0);
+    CPU_MIRROR.fetch_add(upload.bytes, Relaxed);
+    Backing::Releasable(Mutex::new(Releasable { linear, upload: Some(upload) }))
+}

+fn b4_upload_lease_and_partial_snapshot_survive_retirement() {
+    // As the existing host GPU tests, skip where no host is available.
+    let Some(upload) = StagingUpload::new(1024) else { return };
+    let Backing::Releasable(r) = releasable_upload(upload) else { return };
+    let mut r = r.into_inner().unwrap_or_else(|e| e.into_inner());
+    // Two mip spans with distinct data. Simulate an earlier GPU list
+    // retaining its exact copy source, and a metadata-only refresh later.
+    r.linear[..512].fill(0x3a);
+    r.linear[512..].fill(0x71);
+    let lease = r.upload.as_ref().unwrap().clone();
+    assert!(!r.release_if_unleased(true));
+    drop(lease);
+    assert!(!r.release_if_unleased(false));
+    assert!(r.release_if_unleased(true));
+    assert!(r.upload.is_none());
+    // Only the second mip's title pages are readable now.
+    r.linear[512..].fill(0x62);
+    let (restored, remade) = r.restore(1024).expect("recreate upload");
+    assert!(remade);
+    // SAFETY: mapping is live and all 1024 bytes initialized by restore.
+    let bytes = unsafe { std::slice::from_raw_parts(restored.data, 1024) };
+    assert!(bytes[..512].iter().all(|&b| b == 0x3a));
+    assert!(bytes[512..].iter().all(|&b| b == 0x62));
+    assert!(!r.release_if_unleased(true));
+    drop(restored);
+    assert!(r.release_if_unleased(true));
+}
