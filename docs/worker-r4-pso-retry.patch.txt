+/// The SDK validator signs a private copy. Never change the shared translation
+/// or retry a shader the validator rejects; preserve the original binding ABI.
+fn r4_signed(shaders: &[Arc<Shader>]) -> Option<Vec<Vec<u8>>> {
+    use comet_shader::emit::validator::Validator;
+    let mut validator = match Validator::load(Validator::default_path().as_deref()) {
+        Ok(v) => v,
+        Err(error) => {
+            tracing::warn!(%error, ?shaders, "R4 PSO validator unavailable");
+            return None;
+        }
+    };
+    let mut signed = Vec::with_capacity(shaders.len());
+    for shader in shaders {
+        let t = shader.translated()?;
+        tracing::info!(hash = %shader.hash, stage = shader.stage,
+            root_cbvs = ?t.host.root_cbvs, root_srvs = ?t.host.root_srvs,
+            root_uavs = ?t.host.root_uavs, user_dwords = t.host.user_data_dwords(),
+            used = ?t.used, "R4 PSO shader bindings");
+        let mut bytes = t.host.dxil.clone();
+        match validator.validate(&mut bytes) {
+            Ok(()) => tracing::info!(
+                hash = %shader.hash, stage = shader.stage,
+                changed = bytes != t.host.dxil,
+                root_cbvs = ?t.host.root_cbvs, root_srvs = ?t.host.root_srvs,
+                root_uavs = ?t.host.root_uavs,
+                user_dwords = t.host.user_data_dwords(),
+                "R4 PSO DXIL valid"
+            ),
+            Err(error) => {
+                tracing::warn!(hash = %shader.hash, stage = shader.stage, %error, "R4 PSO DXIL invalid");
+                return None;
+            }
+        }
+        signed.push(bytes);
+    }
+    Some(signed)
