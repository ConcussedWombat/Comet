+    /// LICM without speculative memory/descriptor access: only the header
+    /// prefix of a natural loop, with a single unconditional incoming
+    /// preheader. Every lane reaching that edge runs the header once, even
+    /// if the title's loop body takes zero trips. Stop at effects and lane

+    /// have been moved there earlier. Move IDs, not copies: all SSA uses,
+    /// phi edge uses, resource annotations and census tags stay attached.
+    fn p5_licm(&mut self) -> bool {
+        if self.blocks.is_empty() {
+            return false;
+        }

+        let effects = self.effects();

+        for (b, blk) in self.blocks.iter().enumerate() {
+            for &i in blk.phis.iter().chain(&blk.body) {

+            }

+        // same invariant further out when its operands also allow that.
