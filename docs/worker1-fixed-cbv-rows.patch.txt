+/// Match the DXBC header digest of a locally captured target shader.
+/// An all-zero digest is not an identity. The benchmark owner supplies
+/// the two digests from their hash-named captures, never shader bytes.
+fn hair_cbv_digest_selected(digest: [u8; 16], selection: &str) -> bool {
+    const HEX: &[u8; 16] = b"0123456789abcdef";
+    digest != [0; 16]
+        && selection.split(',').map(str::trim).any(|candidate| {
+            candidate.len() == 32
+                && candidate.as_bytes().chunks_exact(2).zip(digest).all(|(hex, b)| {
+                    hex[0].eq_ignore_ascii_case(&HEX[usize::from(b >> 4)])
+                        && hex[1].eq_ignore_ascii_case(&HEX[usize::from(b & 15)])
+                })
+        })
+}
